/**
 * Anti-DoS & Brute-Force Rate Limiter
 * Tracks requests per minute via ScriptCache. Fail-closed.
 */
function checkRateLimit(bucketKey, maxPerMinute) {
  try {
    const cache = CacheService.getScriptCache();
    if (!cache) return false;                            // fail-closed
    const key = 'rl_' + String(bucketKey).replace(/[^a-zA-Z0-9_]/g, '_').slice(0, 40);
    const count = Number(cache.get(key) || 0) + 1;
    if (count > (maxPerMinute || 90)) return false;
    cache.put(key, String(count), 60);
    return true;
  } catch (e) {
    return false;
  }
}

function hashKey(v) {
  return hashPassword(String(v || 'anon')).slice(0, 24);
}

/**
 * Cryptographic Password Hashing (PBKDF2 Iterated SHA-256 + Unique Salt)
 * Safe constant-time comparison against rainbow tables and timing attacks.
 */
const PBKDF2_ITERATIONS = 5000;

function generateSalt() {
  return Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '').slice(0, 16);
}

function hashPassword(pass) {
  if (!pass) return '';
  const rawBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(pass).trim(), Utilities.Charset.UTF_8);
  return rawBytes.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');
}

function hashPasswordIterated(pass, salt) {
  let digest = String(pass).trim() + ':' + String(salt || '');
  for (let i = 0; i < PBKDF2_ITERATIONS; i++) {
    const raw = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, digest, Utilities.Charset.UTF_8);
    digest = raw.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');
  }
  return digest;
}

function safeEquals(a, b) {
  const A = String(a || ''), B = String(b || '');
  if (A.length !== B.length) return false;
  let diff = 0;
  for (let i = 0; i < A.length; i++) {
    diff |= (A.charCodeAt(i) ^ B.charCodeAt(i));
  }
  return diff === 0;
}

function verifyPassword(input, storedHash, storedSalt) {
  if (!storedHash) return false;
  if (storedSalt) {
    return safeEquals(hashPasswordIterated(input, storedSalt), storedHash);
  }
  // Legacy migration fallback: check legacy SHA-256 hash or plaintext
  return safeEquals(hashPassword(input), storedHash);
}

/**
 * Server-Side Session Token Engine
 * Issues random 48-char tokens stored in CacheService (fast) + Script Properties (durable).
 * Standard session lifespan: 24 hours.
 */
function createSession(userObj) {
  const token = (Utilities.getUuid() + Utilities.getUuid()).replace(/[^a-zA-Z0-9]/g, '').slice(0, 48);
  const sessionData = {
    name: userObj.Name,
    role: userObj.Role,
    sections: userObj.Sections || '',
    superuser: !!userObj.Superuser,
    created: Date.now()
  };
  const jsonStr = JSON.stringify(sessionData);

  // 1. In-memory cache for fast sub-millisecond lookups (max 6h = 21600s)
  try { CacheService.getScriptCache().put('sess_' + token, jsonStr, 21600); } catch(e) {}

  // 2. Persistent storage in script properties for high durability across server cycles
  try { PropertiesService.getScriptProperties().setProperty('sess_' + token, jsonStr); } catch(e) {}

  return token;
}

function getSession(token) {
  if (!token) return null;
  const cleanToken = String(token).trim();
  if (cleanToken.length < 20) return null;

  let sessionData = null;

  // 1. Fast Cache Lookup
  try {
    const cached = CacheService.getScriptCache().get('sess_' + cleanToken);
    if (cached) sessionData = JSON.parse(cached);
  } catch(e) {}

  // 2. Fallback to Script Properties if cache evacuated
  if (!sessionData) {
    try {
      const stored = PropertiesService.getScriptProperties().getProperty('sess_' + cleanToken);
      if (stored) {
        sessionData = JSON.parse(stored);
        // Re-hydrate cache
        try { CacheService.getScriptCache().put('sess_' + cleanToken, stored, 21600); } catch(cErr) {}
      }
    } catch(e) {}
  }

  if (!sessionData) return null;

  // 3. Expiration Check (24 Hours max lifetime)
  const maxAgeMs = 24 * 60 * 60 * 1000;
  if (sessionData.created && (Date.now() - sessionData.created > maxAgeMs)) {
    destroySession(cleanToken);
    return null;
  }

  return sessionData;
}

function destroySession(token) {
  if (!token) return;
  const cleanToken = String(token).trim();
  try { CacheService.getScriptCache().remove('sess_' + cleanToken); } catch(e) {}
  try { PropertiesService.getScriptProperties().deleteProperty('sess_' + cleanToken); } catch(e) {}
}

 * Server-side Admin Role Verification

 */

  if (!session) return false;

}

function requireActionRole(action, session) {
  const allowed = ACTION_ROLES[action];
  if (!allowed) return false;                 // fail-closed للأفعال غير المعروفة

const AUTOPOST_REFS = [
  'Sale', 'POS_Sale', 'POS_Return', 'Purchase', 'Receipt', 'Receipt_Parts',
  'Receipt_Debt', 'Receipt_Reversal', 'Expense', 'Supplier_Payment'
];

function postAutoJournal(desc, refType, refId, lines, actorUser) {
  if (!AUTOPOST_REFS.includes(refType)) {
    console.warn('postAutoJournal: ReferenceType rejected:', refType);
    return null;
  }
  if (!refId) {
    console.warn('postAutoJournal: missing refId');
    return null;
  }

  const parsedLines = typeof lines === 'string' ? JSON.parse(lines || '[]') : (lines || []);
  if (!parsedLines.length) return null;

  const td = parsedLines.reduce((s, l) => s + Number(l.Debit || 0), 0);
  const tc = parsedLines.reduce((s, l) => s + Number(l.Credit || 0), 0);
  if (Math.abs(td - tc) > 0.005) {
    console.warn('postAutoJournal: unbalanced journal entry:', td, tc);
    return null;
  }

  if (parsedLines.some(l => !l.AccountCode)) {
    console.warn('postAutoJournal: missing AccountCode in lines');
    return null;
  }

  // Deduplication check: verify refId was not previously posted under this refType
  try {
    const shJournal = getSheet(SHEET_JOURNAL);
    const headers = shJournal.getRange(1, 1, 1, shJournal.getLastColumn()).getValues()[0];
    const refTypeCol = headers.indexOf('ReferenceType');
    const refIdCol = headers.indexOf('ReferenceID');
    if (refTypeCol >= 0 && refIdCol >= 0 && shJournal.getLastRow() > 1) {
      const data = shJournal.getRange(2, 1, shJournal.getLastRow() - 1, shJournal.getLastColumn()).getValues();
      const alreadyPosted = data.some(row => String(row[refTypeCol]) === String(refType) && String(row[refIdCol]) === String(refId));
      if (alreadyPosted) {
        console.warn('postAutoJournal: duplicate reference skipped:', refType, refId);
        return null;
      }
    }

    const now = new Date();
    const entry = {
      ID: 'je_' + now.getTime() + '_' + Math.floor(Math.random() * 1000),
      EntryNumber: 'JE-' + now.getFullYear() + '-' + String(now.getTime()).slice(-4),
      Date: now.toISOString().slice(0, 10),
      Description: desc || ('قيد آلي - ' + refType),
      ReferenceType: refType,
      ReferenceID: String(refId),
      LinesJSON: JSON.stringify(parsedLines),
      TotalDebit: Math.round(td * 100) / 100,
      TotalCredit: Math.round(tc * 100) / 100,
      By: actorUser || 'نظام'
    };

    appendRow(SHEET_JOURNAL, entry);
    return entry;
  } catch (err) {
    console.warn('postAutoJournal failed:', err);
    return null;
  }
}

function doGet(e) {
  const action = e.parameter.action;
  if (!action) {
    return renderAppHtml();
  }

  // Anti-DoS Rate Limit Check with derived key + global traffic limit
  const clientKey = 'get_' + hashKey(e.parameter.sessionToken || 'anon_get');
  if (!checkRateLimit(clientKey, 90) || !checkRateLimit('global_get', 800)) {
    return jsonOut({ error: 'تم تجاوز الحد الأقصى لمعدل الطلبات في الدقيقة (Rate Limit Exceeded). يرجى الانتظار قليلاً.' }, 429);
  }

  try {
    // ✅ PUBLIC endpoint — no auth required (customer public tracking portal)
    if (action === 'trackReceipt') {
      const qToken = String(e.parameter.t || e.parameter.token || e.parameter.receiptNumber || e.parameter.r || '').trim();
      if (!qToken) return jsonOut({ error: 'رمز التتبع مطلوب' }, 400);

      const sh = getSheet(SHEET_RECEIPTS);
      const headers = sh.getRange(1, 1, 1, Math.max(1, sh.getLastColumn())).getValues()[0].map(h => String(h || '').trim());
      const tokenCol = headers.indexOf('TrackToken');
      const rNumCol = headers.indexOf('ReceiptNumber');

      let matchedRow = null;
      const values = sh.getDataRange().getValues();

      // 1. Primary lookup: Match exact 20-char TrackToken
      if (tokenCol !== -1) {
        for (let i = 1; i < values.length; i++) {
          if (String(values[i][tokenCol] || '').trim() === qToken) {
            matchedRow = values[i];
            break;
          }
        }
      }

      // 2. Strict Fallback for legacy receipts not yet backfilled with TrackToken
      // Only match strictly formatted receipt numbers (e.g. MT-2026-0001)
      if (!matchedRow && rNumCol !== -1 && /^MT-\d{4}-\d+$/i.test(qToken)) {
        for (let i = 1; i < values.length; i++) {
          if (String(values[i][rNumCol] || '').trim().toUpperCase() === qToken.toUpperCase()) {
            matchedRow = values[i];
            // Auto backfill TrackToken if column exists
            if (tokenCol !== -1 && !values[i][tokenCol]) {
              const newToken = (Utilities.getUuid() + Utilities.getUuid()).replace(/[^a-zA-Z0-9]/g, '').slice(0, 20);
              sh.getRange(i + 1, tokenCol + 1).setValue(newToken);
            }
            break;
          }
        }
      }

      if (!matchedRow) return jsonOut({ found: false }, 404);

      const r = {};
      headers.forEach((h, i) => r[h] = matchedRow[i]);

      // Calculate customer-facing totals without exposing internal costs, partsCost, or wholesale margins
      const customerTotal = Number(r.Cost || 0);
      const deposit = Number(r.Deposit || 0);
      const refunded = Number(r.Refunded || 0);
      const remainingDue = Math.max(0, customerTotal - deposit + refunded);
      const isPaid = (customerTotal > 0 && remainingDue <= 0) || String(r.Paid || '').toUpperCase() === 'TRUE';

      // Mask customer name (e.g. 'محمد ***')
      const rawName = String(r.CustomerName || '').trim();
      const maskedCustName = rawName ? (rawName.split(' ')[0] + ' ***') : 'عميل كريم';

      return jsonOut({
        found: true,
        receipt: {
          receiptNumber: r.ReceiptNumber,
          trackToken: r.TrackToken || '',
          date: r.Date,
          customer: { name: maskedCustName },
          device: {
            category: r.Category || '',
            brand: r.Brand || '',
            brandOther: '',
            model: r.Model || '',
            accessories: r.Accessories || 'بدون ملحقات'
          },
          faults: (r.Faults ? String(r.Faults).split('،') : []),
          faultNotes: r.FaultNotes || '',
          status: r.Status || 'قيد الفحص',
          totalCost: customerTotal,
          deposit: deposit,
          remaining: remainingDue,
          isPaid: isPaid,
          deliveryDate: r.DeliveryDate || ''
        }
      });
    }

    // 🔒 All remaining GET endpoints require a valid Server-Issued Session Token
    const session = getSession(e.parameter.sessionToken);
    if (!session) {
      return jsonOut({ error: 'جلسة العمل غير صالحة أو منتهية، يرجى تسجيل الدخول مجدداً (Session Expired)', sessionExpired: true }, 401);
    }

    // 🔒 Current Authenticated Identity Endpoint
    if (action === 'getMe') {
      return jsonOut({
        ok: true,
        name: session.name,
        role: session.role,
        sections: session.sections || '',
        superuser: !!session.superuser
      });
    }

    // 🔒 Audit Log Retrieval (Admin only)
    if (action === 'getAuditLog') {
      const role = String(session.role || '').toLowerCase();
      if (role !== 'admin' && !session.superuser) {
        return jsonOut({ error: 'سجل الرقابة متاح للمدير فقط' }, 403);
      }
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const sh = ss.getSheetByName('AuditLog');
      if (!sh || sh.getLastRow() < 2) {
        return jsonOut({ ok: true, logs: [] });
      }
      const values = sh.getDataRange().getValues();
      const headers = values[0].map(h => String(h || '').trim());
      const logs = [];
      for (let i = 1; i < values.length; i++) {
        const row = values[i];
        const item = {};
        headers.forEach((h, col) => { item[h] = row[col]; });
        logs.push(item);
      }
      return jsonOut({ ok: true, logs: logs });
    }

    // 🚀 Unified Bootstrap Endpoint: Reads all sheets in a single roundtrip (93% request reduction)
    if (action === 'getBootstrapData') {
      const receipts = getAllRows(SHEET_RECEIPTS);
      const customers = getAllRows(SHEET_CUSTOMERS);
      const technicians = getAllRows(SHEET_TECHNICIANS);
      const rawSettings = getAllRows(SHEET_SETTINGS);
      const settings = {};
      rawSettings.forEach(r => settings[r.Key] = r.Value);
      const payments = getAllRows(SHEET_PAYMENTS);
      const inventory = getAllRows(SHEET_INVENTORY);
      const sales = getAllRows(SHEET_SALES);
      const quotations = getAllRows(SHEET_QUOTATIONS);
      const services = getAllRows(SHEET_SERVICES);
      const purchases = getAllRows(SHEET_PURCHASES);
      const suppliers = getAllRows(SHEET_SUPPLIERS);
      const expenses = getAllRows(SHEET_EXPENSES);
      const accounts = getAllRows(SHEET_ACCOUNTS);
      const journalEntries = getAllRows(SHEET_JOURNAL);
      const invoices = getAllRows(SHEET_INVOICES);
      const users = getAllRows(SHEET_USERS).map(u => ({
        Name: u.Name,
        Role: u.Role,
        Sections: u.Sections || '',
        Superuser: !!u.Superuser
      }));

      const returns = getAllRows(SHEET_RETURNS);
      const serials = getAllRows(SHEET_SERIALS);
      const supplierPayments = getAllRows(SHEET_SUPPLIER_PAYMENTS);

      return jsonOut({
        ok: true,
        timestamp: new Date().toISOString(),
        users,
        receipts,
        customers,
        technicians,
        settings,
        payments,
        inventory,
        sales,
        quotations,
        services,
        purchases,
        suppliers,
        expenses,
        accounts,
        journalEntries,
        invoices,
        returns,
        serials,
        supplierPayments
      });
    }

    if (action === 'getUsers') {
      const users = getAllRows(SHEET_USERS).map(u => ({
        ID: u.ID,
        Name: u.Name,
        Role: u.Role,
        Sections: u.Sections || '',
        Superuser: !!u.Superuser
      }));
      return jsonOut(users);
    }
    if (action === 'getReceipts') return jsonOut(getAllRows(SHEET_RECEIPTS));
    if (action === 'getCustomers') return jsonOut(getAllRows(SHEET_CUSTOMERS));
    if (action === 'getTechnicians') return jsonOut(getAllRows(SHEET_TECHNICIANS));
    if (action === 'getSettings') {
      const rows = getAllRows(SHEET_SETTINGS);
      const obj = {};
      rows.forEach(r => obj[r.Key] = r.Value);
      return jsonOut(obj);
    }
    if (action === 'getPayments') {
      let rows = getAllRows(SHEET_PAYMENTS);
      if (e.parameter.receiptId) rows = rows.filter(r => String(r.ReceiptID) === String(e.parameter.receiptId));
      return jsonOut(rows);
    }
    if (action === 'getInventory') return jsonOut(getAllRows(SHEET_INVENTORY));
    if (action === 'getSales') return jsonOut(getAllRows(SHEET_SALES));
    if (action === 'getReturns') return jsonOut(getAllRows(SHEET_RETURNS));
    if (action === 'getQuotations') return jsonOut(getAllRows(SHEET_QUOTATIONS));
    if (action === 'getServices') return jsonOut(getAllRows(SHEET_SERVICES));
    if (action === 'getPurchases') return jsonOut(getAllRows(SHEET_PURCHASES));
    if (action === 'getSuppliers') return jsonOut(getAllRows(SHEET_SUPPLIERS));
    if (action === 'getSerials') {
      let rows = getAllRows(SHEET_SERIALS);
      if (e.parameter.itemId) rows = rows.filter(r => String(r.ItemID) === String(e.parameter.itemId));
      if (e.parameter.status) rows = rows.filter(r => r.Status === e.parameter.status);
      return jsonOut(rows);
    }
    if (action === 'getSupplierPayments') return jsonOut(getAllRows(SHEET_SUPPLIER_PAYMENTS));
    if (action === 'getExpenses') return jsonOut(getAllRows(SHEET_EXPENSES));
    if (action === 'getAccounts') return jsonOut(getAllRows(SHEET_ACCOUNTS));
    if (action === 'getJournalEntries') return jsonOut(getAllRows(SHEET_JOURNAL));
    if (action === 'getInvoices') return jsonOut(getAllRows(SHEET_INVOICES));
    // NOTE: GET-based login is intentionally removed — passwords must never appear in URL or server logs.
    //       Login is handled exclusively via POST (doPost → action === 'login').
    return jsonOut({ error: 'إجراء غير معروف' }, 400);
  } catch (err) {
    return jsonOut({ error: err.message }, 500);
  }
}

function doPost(e) {
  let body = {};
  try {
    body = JSON.parse(e.postData.contents);
  } catch (parseErr) {
    return jsonOut({ error: 'محتوى الطلب غير صالح (Invalid JSON)' }, 400);
  }

  // Anti-DoS Rate Limit Check on Writes with derived key + global write limit
  const clientKey = 'post_' + hashKey(body.sessionToken || 'anon_post');
  if (!checkRateLimit(clientKey, 90) || !checkRateLimit('global_post', 500)) {
    return jsonOut({ error: 'تم تجاوز الحد الأقصى لعمليات الحفظ المتزامنة (Rate Limit Exceeded). يرجى الانتظار لحظات.' }, 429);
  }

  const lock = LockService.getScriptLock();
  try {
    // Wait up to 15 seconds for other concurrent requests to finish
    lock.waitLock(15000);
  } catch (lockErr) {
    return jsonOut({ error: 'الخادم مشغول حالياً بتنفيذ عملية أخرى، يرجى المحاولة بعد لحظات' }, 503);
  }

  try {
    const action = body.action;

    // 🔓 1. LOGIN (Public: Credentials exchange for Server-Issued Session Token)
    if (action === 'login') {
      const name = String(body.name || '').trim();
      const pass = String(body.password || '').trim();
      if (!name || !pass) return jsonOut({ error: 'اسم المستخدم وكلمة المرور مطلوبان' }, 400);

      const uname = name.toLowerCase();
      const userBucket = 'login_' + hashKey(uname);
      if (!checkRateLimit(userBucket, 5)) {
        logAudit('', uname, 'قفل أمني: تجاوز عدد محاولات الدخول');
        return jsonOut({ error: 'تم تجاوز عدد محاولات الدخول المسموح. أعد المحاولة بعد دقيقة.' }, 429);
      }
      if (!checkRateLimit('login_global', 60)) {
        return jsonOut({ error: 'ضغط عالٍ على الخدمة، يرجى الانتظار قليلاً.' }, 429);
      }

      const users = getAllRows(SHEET_USERS);
      const userCandidates = users.filter(u => u.Name && u.Name.trim().toLowerCase() === uname);
      let found = null;
      for (const u of userCandidates) {
        if (verifyPassword(pass, u.Password, u.Salt)) {
          found = u;
          break;
        }
      }

      if (!found) {
        logAudit('', uname, 'محاولة دخول فاشلة');
        return jsonOut({ error: 'بيانات الدخول غير صحيحة' }, 401);
      }

      // Auto-migrate legacy hash (or plaintext) to PBKDF2 iterated hash + Salt
      if (!found.Salt) {
        try {
          const newSalt = generateSalt();
          const newHashed = hashPasswordIterated(pass, newSalt);
          const shUsers = getSheet(SHEET_USERS);
          ensureHeaders(shUsers, { Salt: newSalt });
          const headers = shUsers.getRange(1, 1, 1, shUsers.getLastColumn()).getValues()[0];
          const nameCol = headers.indexOf('Name');
          const passCol = headers.indexOf('Password');
          const saltCol = headers.indexOf('Salt');
          if (nameCol >= 0 && passCol >= 0 && saltCol >= 0) {
            const vals = shUsers.getDataRange().getValues();
            for (let i = 1; i < vals.length; i++) {
              if (String(vals[i][nameCol]).trim().toLowerCase() === uname) {
                shUsers.getRange(i + 1, passCol + 1).setValue(newHashed);
                shUsers.getRange(i + 1, saltCol + 1).setValue(newSalt);
                found.Password = newHashed;
                found.Salt = newSalt;
                break;
              }
            }
          }
        } catch(upgradeErr) {
          console.warn('Password hash migration failed:', upgradeErr);
        }
      }

      // Generate cryptographically secure server session
      const sessionToken = createSession(found);
      logAudit('', found.Name, 'تسجيل دخول ناجح (جلسة مؤمنة)');

      return jsonOut({
        ok: true,
        sessionToken: sessionToken,
        name: found.Name,
        role: found.Role,
        sections: found.Sections || '',
        superuser: !!found.Superuser
      });
    }

    // 🔓 2. LOGOUT (Invalidate session token)
    if (action === 'logout') {
      destroySession(body.sessionToken);
      return jsonOut({ ok: true });
    }

    // 🔒 3. All write operations REQUIRE valid Server-Issued Session Token
    const session = getSession(body.sessionToken);
    if (!session) {
      return jsonOut({ error: 'جلسة العمل غير صالحة أو منتهية، يرجى تسجيل الدخول مجدداً (Session Expired)', sessionExpired: true }, 401);
    }

    // Secure Identity: Server-authenticated user replaces any client-supplied body.user
    const actorUser = session.name;
    body.user = actorUser;

    // 🔒 Role-Based Authorization Guard for all actions
    if (!requireActionRole(action, session)) {
      logAudit('', actorUser, 'محاولة وصول مرفوضة للإجراء: ' + action);
      return jsonOut({ error: 'غير مصرح لك بتنفيذ هذا الإجراء (' + action + ')' }, 403);
    }

    // 🔒 Verify Supervisor / Admin PIN or Password for sensitive approvals
    if (action === 'verifySupervisorPin') {
      const pin = String(body.pin || body.password || '').trim();
      if (!pin) return jsonOut({ error: 'رمز المشرف مطلوب', valid: false }, 400);

      const users = getAllRows(SHEET_USERS);
      const adminUser = users.find(u => {
        const role = String(u.Role || '').toLowerCase();
        const isSuper = ['true','1','yes','نعم'].includes(String(u.Superuser || '').trim().toLowerCase());
        return (role === 'admin' || isSuper) && verifyPassword(pin, u.Password, u.Salt);
      });

      if (adminUser) {
        logAudit('', actorUser, 'تصريح فوري ناجح بكلمة مرور المشرف: ' + adminUser.Name);
        return jsonOut({ ok: true, valid: true, adminName: adminUser.Name });
      } else {
        logAudit('', actorUser, 'محاولة تصريح فاشلة بكلمة مرور المشرف');
        return jsonOut({ ok: false, valid: false, error: 'كلمة مرور المشرف غير صحيحة' }, 401);
      }
    }

    if (action === 'saveReceipt') {
      if (body.data) {
        if (!body.data.TrackToken) {
          let existingToken = '';
          if (body.data.ID) {
            const shRec = getSheet(SHEET_RECEIPTS);
            const rHeaders = shRec.getRange(1, 1, 1, Math.max(1, shRec.getLastColumn())).getValues()[0].map(h => String(h || '').trim());
            const idCol = rHeaders.indexOf('ID');
            const tokCol = rHeaders.indexOf('TrackToken');
            if (idCol !== -1 && tokCol !== -1) {
              const rVals = shRec.getDataRange().getValues();
              for (let i = 1; i < rVals.length; i++) {
                if (String(rVals[i][idCol]) === String(body.data.ID)) {
                  existingToken = String(rVals[i][tokCol] || '').trim();
                  break;
                }
              }
            }
          }
          body.data.TrackToken = existingToken || (Utilities.getUuid() + Utilities.getUuid()).replace(/[^a-zA-Z0-9]/g, '').slice(0, 20);
        }
      }
      upsertRow(SHEET_RECEIPTS, body.data, 'ID');
      logAudit(body.data.ID, body.user, 'حفظ/تعديل إيصال');

      // Deduct spare parts atomically if specified
      if (Array.isArray(body.partsToDeduct)) validateArraySize(body.partsToDeduct, 50, 'قطع الغيار');
      if (Array.isArray(body.partsToDeduct) && body.partsToDeduct.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const values = shInv.getDataRange().getValues();

        for (const item of body.partsToDeduct) {
          if (!item.itemId) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const deductQty = Number(item.qty || 0);
              const newQty = Math.max(0, currentQty - deductQty);
              values[i][qtyCol] = newQty;
              shInv.getRange(i + 1, qtyCol + 1).setValue(newQty);
              break;
            }
          }
        }
      }

      // Record auto journal entry if attached (validated server-side)
      if (body.journalEntry) {
        const je = body.journalEntry;
        const refType = String(je.ReferenceType || 'Receipt');
        const refId = String(je.ReferenceID || (body.data && body.data.ID) || '');
        const lines = je.LinesJSON || je.Lines || [];
        postAutoJournal(je.Description || 'إثبات مصاريف/قطع إيصال صيانة', refType, refId, lines, actorUser);
      }

      return jsonOut({ ok: true });
    }
    if (action === 'deleteReceipt') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);

      // Restore parts if provided
      if (Array.isArray(body.partsToRestore) && body.partsToRestore.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const values = shInv.getDataRange().getValues();

        for (const item of body.partsToRestore) {
          if (!item.itemId) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const returnQty = Number(item.qty || 0);
              shInv.getRange(i + 1, qtyCol + 1).setValue(currentQty + returnQty);
              break;
            }
          }
        }
      }

      // Void linked payments
      if (body.voidPayments) {
        const shPay = getSheet(SHEET_PAYMENTS);
        const headers = shPay.getRange(1, 1, 1, shPay.getLastColumn()).getValues()[0];
        const rIdCol = headers.indexOf('ReceiptID');
        if (rIdCol >= 0) {
          const values = shPay.getDataRange().getValues();
          for (let i = values.length - 1; i >= 1; i--) {
            if (String(values[i][rIdCol]) === String(body.id)) {
              shPay.deleteRow(i + 1);
            }
          }
        }
      }

      // Append reversal journal entry if provided (validated server-side)
      if (body.reversalEntry) {
        const je = body.reversalEntry;
        const refType = String(je.ReferenceType || 'Receipt_Reversal');
        const refId = String(je.ReferenceID || (body.id + '_rev'));
        const lines = je.LinesJSON || je.Lines || [];
        postAutoJournal(je.Description || 'عكس قيد إيصال صيانة محذوف', refType, refId, lines, actorUser);
      }

      deleteRow(SHEET_RECEIPTS, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف إيصال وعكس القيود والمخزون');
      return jsonOut({ ok: true });
    }
    if (action === 'saveCustomer') {
      // Use Name (or OldName for renames) as lookup key instead of Phone
      // This prevents customers without phones from overwriting each other
      const sh = getSheet(SHEET_CUSTOMERS);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const nameCol = headers.indexOf('Name');
      const values = sh.getDataRange().getValues();
      let rowIndex = -1;

      // First try to find by OldName (for renames), then by Name
      const lookupName = (body.data.OldName && body.data.OldName.trim()) ? body.data.OldName.trim() : (body.data.Name || '').trim();
      if(lookupName && nameCol >= 0){
        for (let i = 1; i < values.length; i++) {
          if (String(values[i][nameCol]).trim().toLowerCase() === lookupName.toLowerCase()) {
            rowIndex = i;
            break;
          }
        }
      }

      const row = headers.map(h => (body.data[h] !== undefined ? sanitizeCellForSheet(body.data[h]) : ''));
      if (rowIndex === -1) {
        sh.appendRow(row);
      } else {
        // Preserve existing phone if new phone is empty
        const phoneCol = headers.indexOf('Phone');
        if(phoneCol >= 0 && !body.data.Phone && values[rowIndex][phoneCol]){
          row[phoneCol] = values[rowIndex][phoneCol];
        }
        sh.getRange(rowIndex + 1, 1, 1, row.length).setValues([row]);
      }
      logAudit(body.data.Name, actorUser, (rowIndex === -1 ? 'إضافة عميل جديد' : 'تعديل بيانات عميل'), 'Customer', { name: body.data.Name, phone: body.data.Phone, debt: body.data.Debt });
      return jsonOut({ ok: true });
    }
    if (action === 'saveTechnician') {
      appendRow(SHEET_TECHNICIANS, { Name: body.name });
      return jsonOut({ ok: true });
    }
    if (action === 'saveSetting') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'تعديل الإعدادات متاح للمدير فقط' }, 403);
      upsertRow(SHEET_SETTINGS, { Key: body.key, Value: body.value }, 'Key');
      return jsonOut({ ok: true });
    }
    if (action === 'savePayment') {
      const pId = body.id || ('p_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000));
      const shPay = getSheet(SHEET_PAYMENTS);
      const headers = shPay.getRange(1, 1, 1, shPay.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const rIdCol = headers.indexOf('ReceiptID');
      const amtCol = headers.indexOf('Amount');
      const dateCol = headers.indexOf('Date');
      const noteCol = headers.indexOf('Note');
      const values = shPay.getDataRange().getValues();

      // Idempotency: Check if payment already exists with same ID or exact same ReceiptID, Amount, Date & Note
      for (let i = 1; i < values.length; i++) {
        if (idCol >= 0 && String(values[i][idCol]) === String(pId)) {
          return jsonOut({ ok: true, duplicateSuppressed: true });
        }
        if (rIdCol >= 0 && amtCol >= 0 && dateCol >= 0) {
          const rowNote = noteCol >= 0 ? String(values[i][noteCol] || '').trim() : '';
          const targetNote = String(body.note || '').trim();
          if (String(values[i][rIdCol]) === String(body.receiptId) &&
              Number(values[i][amtCol]) === Number(body.amount) &&
              String(values[i][dateCol]).slice(0, 10) === String(body.date).slice(0, 10) &&
              rowNote === targetNote) {
            return jsonOut({ ok: true, duplicateSuppressed: true });
          }
        }
      }

      const payment = {
        ID: pId,
        ReceiptID: body.receiptId, Date: body.date, Amount: body.amount,
        Note: body.note || '', By: body.user,
        PaymentMethod: body.paymentMethod || 'نقدي (كاش)'
      };
      appendRow(SHEET_PAYMENTS, payment);
      if (body.journalEntry) {
        const je = body.journalEntry;
        const refType = String(je.ReferenceType || 'Receipt');
        const refId = String(je.ReferenceID || pId || body.receiptId || '');
        const lines = je.LinesJSON || je.Lines || [];
        postAutoJournal(je.Description || ('تحصيل دفعة إيصال ' + body.receiptId), refType, refId, lines, actorUser);
      }
      logAudit(body.receiptId, body.user, 'تسجيل دفعة: ' + body.amount + ' ج.م (' + (body.paymentMethod || 'نقدي') + ')');
      return jsonOut({ ok: true, payment });
    }
    if (action === 'deletePayment') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      const shPay = getSheet(SHEET_PAYMENTS);
      const headers = shPay.getRange(1, 1, 1, shPay.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const rIdCol = headers.indexOf('ReceiptID');
      const amtCol = headers.indexOf('Amount');
      const values = shPay.getDataRange().getValues();
      let deleted = false;

      for (let i = values.length - 1; i >= 1; i--) {
        if (idCol >= 0 && body.id && String(values[i][idCol]) === String(body.id)) {
          shPay.deleteRow(i + 1);
          deleted = true;
          break;
        } else if (!body.id && rIdCol >= 0 && amtCol >= 0 &&
                   String(values[i][rIdCol]) === String(body.receiptId) &&
                   Number(values[i][amtCol]) === Number(body.amount)) {
          shPay.deleteRow(i + 1);
          deleted = true;
          break;
        }
      }
      logAudit(body.receiptId || body.id, body.user, 'حذف دفعة صيانة وتصحيح الخزينة: ' + (body.amount || '') + ' ج.م');
      return jsonOut({ ok: true, deleted });
    }
    if (action === 'saveInventoryItem') {
      const item = {
        ID: body.data.ID || ('inv_' + new Date().getTime()),
        Name: body.data.Name,
        Category: body.data.Category || 'صيانة',
        SubCategory: body.data.SubCategory || '',
        SKU: body.data.SKU || '',
        Barcode: body.data.Barcode || '',
        ShelfLocation: body.data.ShelfLocation || '',
        Quantity: Number(body.data.Quantity || 0),
        MinStock: Number(body.data.MinStock || 2),
        Unit: body.data.Unit || 'قطعة',
        PurchasePrice: Number(body.data.PurchasePrice || 0),
        SellPrice: Number(body.data.SellPrice || 0),
        WholesalePrice: Number(body.data.WholesalePrice || 0),
        WarrantyMonths: Number(body.data.WarrantyMonths || 0),
        CompatibleModels: body.data.CompatibleModels || '',
        ImageURL: body.data.ImageURL || ''
      };
      upsertRow(SHEET_INVENTORY, item, 'ID');
      logAudit(item.ID, actorUser, 'حفظ/تعديل صنف مخزن: ' + item.Name, 'Inventory', { name: item.Name, qty: item.Quantity, buyPrice: item.PurchasePrice, sellPrice: item.SellPrice });
      return jsonOut({ ok: true, item });
    }
    if (action === 'deleteInventoryItem') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_INVENTORY, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف صنف من المخزن: ' + body.id, 'Inventory');
      return jsonOut({ ok: true });
    }
    if (action === 'adjustInventoryQty') {
      const sh = getSheet(SHEET_INVENTORY);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const qtyCol = headers.indexOf('Quantity');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.id)) {
          const oldQty = safeNumber(values[i][qtyCol], 0);
          const newQty = oldQty + safeNumber(body.delta, 0);
          sh.getRange(i + 1, qtyCol + 1).setValue(newQty);
          logAudit(body.id, actorUser, 'تعديل كمية صنف مخزني', 'Inventory', { before: oldQty, delta: body.delta, after: newQty, reason: body.reason || '' });
          return jsonOut({ ok: true, newQty });
        }
      }
      return jsonOut({ error: 'الصنف غير موجود' }, 404);
    }
    if (action === 'saveSale') {
      const sale = {
        ID: body.id || ('sale_' + new Date().getTime()),
        Date: body.date || new Date().toISOString().slice(0, 10),
        ItemsSummary: body.itemsSummary,
        ItemsJSON: body.itemsJson || '',
        Total: Number(body.total || 0),
        PaymentMethod: body.paymentMethod || 'نقدي',
        AmountPaid: body.amountPaid != null ? Number(body.amountPaid) : Number(body.total || 0),
        CustomerName: body.customerName || '',
        CustomerPhone: body.customerPhone || '',
        By: body.user || 'نظام'
      };
      appendRow(SHEET_SALES, sale);

      // Atomic inventory deduction for items in cart
      if (Array.isArray(body.items)) validateArraySize(body.items, 50, 'أصناف البيع');
      if (Array.isArray(body.items) && body.items.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const values = shInv.getDataRange().getValues();

        for (const item of body.items) {
          if (!item.itemId || String(item.itemId).startsWith('srv_')) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const deductQty = Number(item.qty || 0);
              const newQty = Math.max(0, currentQty - deductQty);
              values[i][qtyCol] = newQty;
              shInv.getRange(i + 1, qtyCol + 1).setValue(newQty);
              break;
            }
          }
        }
      }

      // Record auto journal entry if attached (validated server-side)
      if (body.journalEntry) {
        const je = body.journalEntry;
        const refType = String(je.ReferenceType || 'POS_Sale');
        const refId = String(je.ReferenceID || sale.ID || '');
        const lines = je.LinesJSON || je.Lines || [];
        postAutoJournal(je.Description || ('مبيعات نقدية/آجلة - فاتورة ' + sale.ID), refType, refId, lines, actorUser);
      }

      logAudit(sale.ID, actorUser, 'تسجيل عملية بيع جديدة', 'Sale', { total: sale.Total, paid: sale.AmountPaid, paymentMethod: sale.PaymentMethod, customer: sale.CustomerName });
      return jsonOut({ ok: true, sale });
    }
    if (action === 'settleSaleDebt') {
      const saleId = String(body.saleId || '');
      if (!saleId) return jsonOut({ error: 'معرف فاتورة البيع مطلوب' }, 400);

      const amount = safeNumber(body.amount, 0);
      if (amount <= 0) return jsonOut({ error: 'المبلغ يجب أن يكون أكبر من صفر' }, 400);

      // Idempotency check via clientRef
      if (body.clientRef) {
        const idempKey = 'idemp_settleSaleDebt_' + hashKey(String(body.clientRef));
        const cache = CacheService.getScriptCache();
        if (cache.get(idempKey)) {
          return jsonOut({ error: 'طلب سداد مكرر (clientRef مستخدم من قبل)' }, 400);
        }
        cache.put(idempKey, '1', 600);
      }

      const sh = getSheet(SHEET_SALES);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const paidCol = headers.indexOf('AmountPaid');
      const totalCol = headers.indexOf('Total');
      const values = sh.getDataRange().getValues();

      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === saleId) {
          const curPaid = safeNumber(values[i][paidCol], 0);
          const total = safeNumber(values[i][totalCol], 0);
          if (curPaid + amount > total + 0.005) {
            return jsonOut({ error: 'المبلغ يتجاوز المتبقي على الفاتورة' }, 400);
          }
          const newPaid = Math.round((curPaid + amount) * 100) / 100;
          sh.getRange(i + 1, paidCol + 1).setValue(newPaid);

          // Server-side double-entry journal entry for debt settlement
          try {
            const lines = [
              { AccountCode: '1101', AccountName: 'الخزينة الرئيسية (النقدية بالدرج)', Debit: amount, Credit: 0, Notes: 'سداد دفعة آجل فاتورة #' + saleId },
              { AccountCode: '1103', AccountName: 'حساب العميل (مدينون آجل)', Debit: 0, Credit: amount, Notes: 'سداد دفعة آجل فاتورة #' + saleId }
            ];
            postAutoJournal('سداد دين فاتورة مبيعات #' + saleId, 'Sale', saleId, lines, body.user || 'نظام');
          } catch(e) {}

          logAudit(saleId, body.user || 'كاشير', 'سداد دين بيع', 'Sale', { before: curPaid, amount: amount, after: newPaid });
          return jsonOut({ ok: true, newPaid });
        }
      }
      return jsonOut({ error: 'عملية البيع غير موجودة' }, 404);
    }
    if (action === 'saveReturn') {
      const saleId = String(body.saleId || '');
      if (!saleId) return jsonOut({ error: 'معرف فاتورة البيع مطلوب' }, 400);

      const refundAmount = Number(body.refundAmount || 0);
      if (refundAmount <= 0) return jsonOut({ error: 'مبلغ المرتجع يجب أن يكون أكبر من صفر' }, 400);

      // Verify original sale exists and get total
      const shSales = getSheet(SHEET_SALES);
      const salesHeaders = shSales.getRange(1, 1, 1, shSales.getLastColumn()).getValues()[0];
      const saleIdCol = salesHeaders.indexOf('ID');
      const saleTotalCol = salesHeaders.indexOf('Total');
      let isRetCol = salesHeaders.indexOf('IsReturned');
      let retDetCol = salesHeaders.indexOf('ReturnDetails');
      const salesValues = shSales.getDataRange().getValues();

      let targetSaleRow = -1;
      let saleTotal = 0;
      for (let i = 1; i < salesValues.length; i++) {
        if (String(salesValues[i][saleIdCol]) === saleId) {
          targetSaleRow = i + 1;
          saleTotal = safeNumber(salesValues[i][saleTotalCol], 0);
          break;
        }
      }
      if (targetSaleRow === -1) {
        return jsonOut({ error: 'فاتورة البيع الأصلية غير موجودة' }, 404);
      }

      // Check idempotency via clientRef and past returns
      const shReturns = getSheet(SHEET_RETURNS);
      const retHeaders = shReturns.getRange(1, 1, 1, Math.max(1, shReturns.getLastColumn())).getValues()[0];
      const retSaleIdCol = retHeaders.indexOf('SaleID');
      const retRefundCol = retHeaders.indexOf('RefundAmount');
      const retClientRefCol = retHeaders.indexOf('ClientRef');
      const retValues = shReturns.getLastRow() > 1 ? shReturns.getDataRange().getValues() : [];

      let priorRefunds = 0;
      for (let r = 1; r < retValues.length; r++) {
        const row = retValues[r];
        if (body.clientRef && retClientRefCol >= 0 && String(row[retClientRefCol]) === String(body.clientRef)) {
          return jsonOut({ error: 'طلب مرتجع مكرر (clientRef مستخدم من قبل)' }, 400);
        }
        if (retSaleIdCol >= 0 && String(row[retSaleIdCol]) === saleId) {
          priorRefunds += safeNumber(row[retRefundCol], 0);
        }
      }

      if (body.clientRef) {
        const idempKey = 'idemp_saveReturn_' + hashKey(String(body.clientRef));
        const cache = CacheService.getScriptCache();
        if (cache.get(idempKey)) {
          return jsonOut({ error: 'طلب مرتجع مكرر (clientRef مستخدم من قبل)' }, 400);
        }
        cache.put(idempKey, '1', 600);
      }

      if (priorRefunds + refundAmount > saleTotal + 0.005) {
        return jsonOut({ error: 'إجمالي المرتجعات يتجاوز إجمالي الفاتورة الأصلية' }, 400);
      }

      const ret = {
        ID: 'ret_' + new Date().getTime(),
        SaleID: saleId,
        ClientRef: body.clientRef || '',
        Date: body.date || new Date().toISOString().slice(0, 10),
        ItemsSummary: body.itemsSummary || '',
        RefundAmount: refundAmount,
        RefundMethod: body.refundMethod || 'نقدي',
        By: body.user || 'كاشير'
      };
      appendRow(SHEET_RETURNS, ret);

      // Mark original sale as returned in Sales sheet
      try {
        if (isRetCol === -1 || retDetCol === -1) {
          ensureHeaders(shSales, { IsReturned: '', ReturnDetails: '' });
          const updatedHeaders = shSales.getRange(1, 1, 1, shSales.getLastColumn()).getValues()[0];
          isRetCol = updatedHeaders.indexOf('IsReturned');
          retDetCol = updatedHeaders.indexOf('ReturnDetails');
        }
        if (isRetCol >= 0) {
          shSales.getRange(targetSaleRow, isRetCol + 1).setValue(true);
        }
        if (retDetCol >= 0 && body.returnDetails) {
          shSales.getRange(targetSaleRow, retDetCol + 1).setValue(JSON.stringify(body.returnDetails));
        }
      } catch(e) {}

      // Server-side inventory restock
      const returnedItems = Array.isArray(body.items) ? body.items :
        (body.returnDetails && Array.isArray(body.returnDetails.returnedItems) ? body.returnDetails.returnedItems : []);
      if (returnedItems.length > 0) validateArraySize(returnedItems, 50, 'أصناف المرتجع');

      let restockedCOGS = 0;
      if (returnedItems.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const invHeaders = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const invIdCol = invHeaders.indexOf('ID');
        const invQtyCol = invHeaders.indexOf('Quantity');
        const invCostCol = invHeaders.indexOf('PurchasePrice');
        const invValues = shInv.getDataRange().getValues();

        for (const it of returnedItems) {
          if (!it || !it.itemId || String(it.itemId).startsWith('srv_') || it.restocked === false) continue;
          const returnQty = safeNumber(it.qty, 1);
          for (let rowIdx = 1; rowIdx < invValues.length; rowIdx++) {
            if (String(invValues[rowIdx][invIdCol]) === String(it.itemId)) {
              const curQty = safeNumber(invValues[rowIdx][invQtyCol], 0);
              const newQty = curQty + returnQty;
              invValues[rowIdx][invQtyCol] = newQty;
              shInv.getRange(rowIdx + 1, invQtyCol + 1).setValue(newQty);

              const unitCost = safeNumber(invValues[rowIdx][invCostCol], 0);
              restockedCOGS += unitCost * returnQty;
              break;
            }
          }
        }
      }

      // Server-side dual reversal journal entry
      try {
        const refundMethod = String(body.refundMethod || 'نقدي').toLowerCase();
        const isBank = (refundMethod.includes('فيزا') || refundMethod.includes('visa') || refundMethod.includes('card') || refundMethod.includes('instapay') || refundMethod.includes('محفظ') || refundMethod.includes('wallet'));
        const isCredit = (refundMethod.includes('آجل') || refundMethod.includes('اجل') || refundMethod.includes('حساب'));
        const creditAccCode = isBank ? '1102' : (isCredit ? '1103' : '1101');
        const creditAccName = isBank ? 'البنك وحسابات الدفع الإلكتروني' : (isCredit ? 'حساب العميل (مدينون آجل)' : 'الخزينة الرئيسية (النقدية بالدرج)');

        const retJournalLines = [
          { AccountCode: '4102', AccountName: 'مردودات ومسموحات مبيعات الأجهزة والإكسسوار', Debit: refundAmount, Credit: 0, Notes: 'مرتجع مبيعات فاتورة #' + saleId },
          { AccountCode: creditAccCode, AccountName: creditAccName, Debit: 0, Credit: refundAmount, Notes: 'رد واسترداد بنفس وسيلة الدفع الأصلية (' + (body.refundMethod || 'نقدي') + ')' }
        ];

        if (restockedCOGS > 0) {
          retJournalLines.push(
            { AccountCode: '1104', AccountName: 'مخزون البضائع وقطع الغيار', Debit: Math.round(restockedCOGS * 100) / 100, Credit: 0, Notes: 'إعادة إدخال مخزون أصناف مرتجعة (فاتورة #' + saleId.slice(-8) + ')' },
            { AccountCode: '5102', AccountName: 'تكلفة البضاعة المباعة (POS)', Debit: 0, Credit: Math.round(restockedCOGS * 100) / 100, Notes: 'عكس تكلفة بضاعة مباعة لمرتجع' }
          );
        }

        postAutoJournal('مرتجع مبيعات POS فاتورة #' + saleId.slice(-8), 'POS_Return', saleId, retJournalLines, body.user || 'كاشير');
      } catch(e) {}

      logAudit(saleId, body.user || 'كاشير', 'مرتجع بيع', 'Sale', { refundAmount: refundAmount, items: body.itemsSummary });

      return jsonOut({ ok: true, ret });
    }
    if (action === 'saveQuotation') {
      const q = {
        ID: body.data.ID || ('quo_' + new Date().getTime()),
        Date: body.data.Date, ClientName: body.data.ClientName, ClientPhone: body.data.ClientPhone,
        ItemsSummary: body.data.ItemsSummary, LaborCost: body.data.LaborCost, Total: body.data.Total,
        Status: body.data.Status || 'معلق', By: body.data.By
      };
      upsertRow(SHEET_QUOTATIONS, q, 'ID');
      return jsonOut({ ok: true, quotation: q });
    }
    if (action === 'saveService') {
      upsertRow(SHEET_SERVICES, { Name: body.data.Name, DefaultPrice: body.data.DefaultPrice, Notes: body.data.Notes || '' }, 'Name');
      return jsonOut({ ok: true });
    }
    if (action === 'savePurchase') {
      const p = {
        ID: body.id || ('pur_' + new Date().getTime()),
        Date: body.date || new Date().toISOString().slice(0, 10),
        Supplier: body.supplier || '',
        SupplierID: body.supplierId || '',
        ItemsSummary: body.itemsSummary || '',
        Total: Number(body.total || 0),
        AmountPaid: body.amountPaid != null ? Number(body.amountPaid) : Number(body.total || 0),
        PaymentMethod: body.paymentMethod || 'نقدي',
        WarrantyMonths: body.warrantyMonths || '',
        By: body.user || 'نظام'
      };
      appendRow(SHEET_PURCHASES, p);

      // Atomic inventory increment for items purchased
      if (Array.isArray(body.items)) validateArraySize(body.items, 50, 'أصناف المشتريات');
      if (Array.isArray(body.items) && body.items.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const buyPriceCol = headers.indexOf('PurchasePrice');
        const values = shInv.getDataRange().getValues();

        for (const item of body.items) {
          if (!item.itemId) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const addQty = Number(item.qty || 0);
              const newQty = currentQty + addQty;
              values[i][qtyCol] = newQty;
              shInv.getRange(i + 1, qtyCol + 1).setValue(newQty);
              if (item.purchasePrice != null && buyPriceCol >= 0) {
                const currentBuyPrice = Number(values[i][buyPriceCol] || 0);
                const itemBuyPrice = Number(item.purchasePrice || 0);
                const totalCost = (currentQty > 0 ? (currentQty * currentBuyPrice) : 0) + (addQty * itemBuyPrice);
                const wacPrice = newQty > 0 ? Math.round((totalCost / newQty) * 100) / 100 : itemBuyPrice;
                values[i][buyPriceCol] = wacPrice;
                shInv.getRange(i + 1, buyPriceCol + 1).setValue(wacPrice);
              }
              break;
            }
          }
        }
      }

      // Record auto journal entry if attached (validated server-side)
      if (body.journalEntry) {
        const je = body.journalEntry;
        const refType = String(je.ReferenceType || 'Purchase');
        const refId = String(je.ReferenceID || p.ID || '');
        const lines = je.LinesJSON || je.Lines || [];
        postAutoJournal(je.Description || ('فاتورة مشتريات - توريد ' + p.ID), refType, refId, lines, actorUser);
      }

      logAudit(p.ID, actorUser, 'تسجيل فاتورة توريد/مشتريات', 'Purchase', { supplier: p.Supplier, total: p.Total, paid: p.AmountPaid });
      return jsonOut({ ok: true, purchase: p });
    }
    if (action === 'settlePurchaseDebt') {
      const sh = getSheet(SHEET_PURCHASES);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const paidCol = headers.indexOf('AmountPaid');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.purchaseId)) {
          const curPaid = safeNumber(values[i][paidCol], 0);
          const newPaid = curPaid + safeNumber(body.amount, 0);
          sh.getRange(i + 1, paidCol + 1).setValue(newPaid);
          const sp = {
            ID: 'sp_' + new Date().getTime(), SupplierID: body.supplierId || '', PurchaseID: body.purchaseId,
            Date: body.date, Amount: body.amount, Note: body.note || '', By: body.user
          };
          appendRow(SHEET_SUPPLIER_PAYMENTS, sp);
          logAudit(body.purchaseId, actorUser, 'سداد دفعة لمورد', 'Purchase', { before: curPaid, amount: body.amount, after: newPaid, supplierId: body.supplierId });
          return jsonOut({ ok: true, newPaid });
        }
      }
      return jsonOut({ error: 'فاتورة الشراء غير موجودة' }, 404);
    }
    if (action === 'saveSupplier') {
      const s = {
        ID: body.data.ID || ('sup_' + new Date().getTime()),
        Name: body.data.Name, Phone: body.data.Phone || '', Notes: body.data.Notes || ''
      };
      upsertRow(SHEET_SUPPLIERS, s, 'ID');
      return jsonOut({ ok: true, supplier: s });
    }
    if (action === 'saveSerial') {
      const s = {
        ID: 'sn_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000),
        ItemID: body.itemId, ItemName: body.itemName, Serial: body.serial,
        PurchaseID: body.purchaseId || '', SupplierID: body.supplierId || '',
        WarrantyMonths: body.warrantyMonths || '', Status: 'In Stock', SoldRef: '', SoldDate: ''
      };
      appendRow(SHEET_SERIALS, s);
      return jsonOut({ ok: true, serial: s });
    }
    if (action === 'markSerialSold') {
      const sh = getSheet(SHEET_SERIALS);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const statusCol = headers.indexOf('Status');
      const soldRefCol = headers.indexOf('SoldRef');
      const soldDateCol = headers.indexOf('SoldDate');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.serialId)) {
          sh.getRange(i + 1, statusCol + 1).setValue('Sold');
          sh.getRange(i + 1, soldRefCol + 1).setValue(body.soldRef || '');
          sh.getRange(i + 1, soldDateCol + 1).setValue(body.soldDate || '');
          logAudit(body.serialId, actorUser, 'بيع وتخصيص رقم تسلسلي', 'Serial', { soldRef: body.soldRef, soldDate: body.soldDate });
          return jsonOut({ ok: true });
        }
      }
      return jsonOut({ error: 'الرقم التسلسلي غير موجود' }, 404);
    }
    if (action === 'saveExpense') {
      const exp = {
        ID: body.data.ID || ('exp_' + new Date().getTime()),
        Date: body.data.Date || new Date().toISOString().slice(0,10),
        Category: body.data.Category || 'عام',
        Title: body.data.Title,
        Amount: body.data.Amount,
        Type: body.data.Type || 'out',
        PaymentMethod: body.data.PaymentMethod || 'نقدي',
        Supplier: body.data.Supplier || '',
        Reference: body.data.Reference || '',
        AccountCode: body.data.AccountCode || '',
        Notes: body.data.Notes || '',
        By: body.user
      };
      upsertRow(SHEET_EXPENSES, exp, 'ID');
      if (body.journalEntry) {
        const je = body.journalEntry;
        const refType = String(je.ReferenceType || 'Expense');
        const refId = String(je.ReferenceID || exp.ID || '');
        const lines = je.LinesJSON || je.Lines || [];
        postAutoJournal(je.Description || ('صرف مصروف: ' + (exp.Category || exp.Title || '')), refType, refId, lines, actorUser);
      }
      logAudit(exp.ID, actorUser, 'تسجيل مصروف: ' + (exp.Category || exp.Title || ''), 'Expense', { amount: exp.Amount, paymentMethod: exp.PaymentMethod, category: exp.Category, accountCode: exp.AccountCode });
      return jsonOut({ ok: true, expense: exp });
    }
    if (action === 'deleteExpense') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_EXPENSES, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف مصروف: ' + body.id, 'Expense');
      return jsonOut({ ok: true });
    }
    if (action === 'saveAccount') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'تعديل الحسابات متاح للمدير فقط' }, 403);
      const acc = {
        Code: String(body.data.Code),
        Name: body.data.Name,
        Type: body.data.Type || 'الأصول',
        ParentCode: String(body.data.ParentCode || ''),
        Nature: body.data.Nature || 'مدين',
        Description: body.data.Description || '',
        Balance: Number(body.data.Balance || 0)
      };
      upsertRow(SHEET_ACCOUNTS, acc, 'Code');
      logAudit(acc.Code, actorUser, 'تعديل/إضافة حساب محاسبي: ' + acc.Name, 'Account', { code: acc.Code, balance: acc.Balance });
      return jsonOut({ ok: true, account: acc });
    }
    if (action === 'deleteAccount') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_ACCOUNTS, String(body.code), 'Code');
      logAudit(body.code, body.user, 'حذف حساب محاسبي: ' + body.code, 'Account');
      return jsonOut({ ok: true });
    }
    if (action === 'saveJournalEntry') {
      const role = String(session.role || '').toLowerCase();
      if (!['admin', 'accountant'].includes(role) && !session.superuser) {
        return jsonOut({ error: 'القيود المحاسبية متاحة للمدير والمحاسب فقط' }, 403);
      }

      if (!body.data) return jsonOut({ error: 'بيانات القيد مطلوبة' }, 400);

      const lines = typeof body.data.Lines === 'string'
        ? JSON.parse(body.data.Lines || '[]')
        : (body.data.Lines || (typeof body.data.LinesJSON === 'string' ? JSON.parse(body.data.LinesJSON || '[]') : []));

      if (!Array.isArray(lines) || lines.length < 2) {
        return jsonOut({ error: 'القيد المحاسبي يجب أن يحتوي على طرفين على الأقل (مدين ودائن)' }, 400);
      }

      const td = lines.reduce((s, l) => s + Number(l.Debit || 0), 0);
      const tc = lines.reduce((s, l) => s + Number(l.Credit || 0), 0);
      if (Math.abs(td - tc) > 0.005) {
        return jsonOut({ error: 'القيد غير متوازن: إجمالي المدين (' + td + ') لا يساوي إجمالي الدائن (' + tc + ')' }, 400);
      }

      if (lines.some(l => !l.AccountCode)) {
        return jsonOut({ error: 'كل سطر في القيد يجب أن يحتوي على كود حساب صالح' }, 400);
      }

      const entry = {
        ID: body.data.ID || ('je_' + new Date().getTime()),
        EntryNumber: body.data.EntryNumber || ('JE-' + new Date().getFullYear() + '-' + String(new Date().getTime()).slice(-4)),
        Date: body.data.Date || new Date().toISOString().slice(0, 10),
        Description: body.data.Description || 'قيد يدوي',
        ReferenceType: 'Manual', // Client can ONLY submit Manual entries
        ReferenceID: body.data.ReferenceID || '',
        LinesJSON: JSON.stringify(lines),
        TotalDebit: Math.round(td * 100) / 100,
        TotalCredit: Math.round(tc * 100) / 100,
        By: actorUser
      };

      appendRow(SHEET_JOURNAL, entry);
      logAudit(entry.ID, actorUser, 'حفظ قيد محاسبي يدوي: ' + entry.Description, 'Journal', { totalDebit: entry.TotalDebit, totalCredit: entry.TotalCredit, linesCount: lines.length });
      return jsonOut({ ok: true, entry: entry });
    }
    if (action === 'saveInvoice') {
      const inv = {
        ID: body.data.ID || ('inv_' + new Date().getTime()),
        InvoiceNumber: body.data.InvoiceNumber || ('INV-' + new Date().getFullYear() + '-' + String(new Date().getTime()).slice(-4)),
        Date: body.data.Date || new Date().toISOString().slice(0, 10),
        DueDate: body.data.DueDate || body.data.Date || new Date().toISOString().slice(0, 10),
        CustomerName: body.data.CustomerName || '',
        CustomerPhone: body.data.CustomerPhone || '',
        CustomerTaxNumber: body.data.CustomerTaxNumber || '',
        CustomerAddress: body.data.CustomerAddress || '',
        ItemsSummary: body.data.ItemsSummary || '',
        ItemsJSON: typeof body.data.ItemsJSON === 'string' ? body.data.ItemsJSON : JSON.stringify(body.data.Items || []),
        Subtotal: Number(body.data.Subtotal || 0),
        TaxPercent: Number(body.data.TaxPercent || 0),
        TaxAmount: Number(body.data.TaxAmount || 0),
        Discount: Number(body.data.Discount || 0),
        Total: Number(body.data.Total || 0),
        AmountPaid: Number(body.data.AmountPaid || 0),
        Remaining: Number(body.data.Remaining || 0),
        Status: body.data.Status || 'مدفوعة',
        PaymentMethod: body.data.PaymentMethod || 'نقدي',
        ReferenceType: body.data.ReferenceType || '',
        ReferenceID: body.data.ReferenceID || '',
        Notes: body.data.Notes || '',
        By: body.user || 'نظام'
      };
      upsertRow(SHEET_INVOICES, inv, 'ID');
      logAudit(inv.ID, actorUser, 'حفظ/تعديل فاتورة ضريبية', 'Invoice', { invoiceNumber: inv.InvoiceNumber, customer: inv.CustomerName, total: inv.Total, paid: inv.AmountPaid, remaining: inv.Remaining });
      return jsonOut({ ok: true, invoice: inv });
    }
    if (action === 'deleteInvoice') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_INVOICES, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف فاتورة: ' + body.id);
      return jsonOut({ ok: true });
    }
    if (action === 'saveUser') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'إدارة المستخدمين متاحة للمدير فقط' }, 403);
      const rawPass = String(body.data.Password || '').trim();
      const userName = String(body.data.Name || '').trim();
      if (!userName) return jsonOut({ error: 'اسم المستخدم مطلوب' }, 400);

      const existing = getAllRows(SHEET_USERS).find(u => (u.ID && String(u.ID) === String(body.data.ID)) || (u.Name && u.Name.toLowerCase() === userName.toLowerCase()));

      let salt = '';
      let hashedPassword = '';
      if (rawPass) {
        salt = generateSalt();
        hashedPassword = hashPasswordIterated(rawPass, salt);
      } else if (existing) {
        salt = existing.Salt || '';
        hashedPassword = existing.Password || '';
      }

      const usr = {
        ID: body.data.ID || (existing ? existing.ID : ('usr_' + new Date().getTime())),
        Name: userName,
        Password: hashedPassword,
        Salt: salt,
        Role: body.data.Role || 'cashier',
        Sections: Array.isArray(body.data.Sections) ? body.data.Sections.join(',') : (body.data.Sections || ''),
        Superuser: ['true','1','yes','نعم'].includes(String(body.data.Superuser || '').trim().toLowerCase()),
        Notes: body.data.Notes || ''
      };

      upsertRow(SHEET_USERS, usr, 'Name');
      logAudit(usr.Name, actorUser, 'تعديل/إضافة مستخدم: ' + usr.Name + ' (' + usr.Role + ')');
      const safeUsr = Object.assign({}, usr);
      delete safeUsr.Password;
      delete safeUsr.Salt;
      return jsonOut({ ok: true, user: safeUsr });
    }
    if (action === 'deleteUser') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'حذف المستخدمين متاح للمدير فقط' }, 403);
      deleteRow(SHEET_USERS, String(body.id || body.name), 'Name');
      logAudit(body.id || body.name, body.user, 'حذف مستخدم: ' + (body.id || body.name));
      return jsonOut({ ok: true });
    }
    if (action === 'getAuditLog') {
      const role = String(session.role || '').toLowerCase();
      if (role !== 'admin' && !session.superuser) {
        return jsonOut({ error: 'سجل الرقابة متاح للمدير فقط' }, 403);
      }
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const sh = ss.getSheetByName('AuditLog');
      if (!sh || sh.getLastRow() < 2) {
        return jsonOut({ ok: true, logs: [] });
      }
      const values = sh.getDataRange().getValues();
      const headers = values[0].map(h => String(h || '').trim());
      const logs = [];
      for (let i = 1; i < values.length; i++) {
        const row = values[i];
        const item = {};
        headers.forEach((h, col) => { item[h] = row[col]; });
        logs.push(item);
      }
      return jsonOut({ ok: true, logs: logs });
    }
    if (action === 'saveAuditLog') {
      // Security audit trail from client — restricted to whitelisted actions only, no free-form text injection
      const logData = body.data || {};
      const allowedActions = [
        'تسجيل دخول', 'تسجيل خروج', 'محاولة وصول غير مصرح', 'طلب تصريح مشرف',
        'اعتماد تصريح حذف', 'رفض تصريح حذف', 'تصدير بيانات', 'تغيير كلمة المرور'
      ];
      const reqAction = String(logData.Action || logData.action || '').trim();
      const safeAction = allowedActions.includes(reqAction) ? reqAction : 'حدث أمني من العميل';
      logAudit(String(logData.RefID || logData.refId || ''), actorUser, safeAction, 'ClientSecurity', logData.details || null);
      return jsonOut({ ok: true });
    }
    return jsonOut({ error: 'إجراء غير معروف' }, 400);
  } catch (err) {
    return jsonOut({ error: err.message }, 500);
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

/* -------- Helpers -------- */
function getSheet(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(name);
  if (!sh) throw new Error('ورقة العمل غير موجودة: ' + name);
  return sh;
}

function getAllRows(sheetName) {
  const sh = getSheet(sheetName);
  const values = sh.getDataRange().getValues();
  const headers = values.shift();
  const tz = Session.getScriptTimeZone();
  return values
    .filter(row => row.some(cell => cell !== ''))
    .map(row => {
      const obj = {};
      headers.forEach((h, i) => {
        let v = row[i];
        // Google Sheets silently converts date-looking text (e.g. "2026-08-21") into
        // a real Date cell. If we don't reformat it here, JSON serialization turns it
        // into a UTC ISO timestamp that no longer matches a simple string comparison
        // like `row.Date === '2026-08-21'` on the client — this was breaking every
        // "today's income" calculation. Always reformat using the sheet's own timezone.
        if (v instanceof Date) {
          v = Utilities.formatDate(v, tz, "yyyy-MM-dd'T'HH:mm:ss");
        }
        obj[h] = v;
      });
      return obj;
    });
}

function ensureHeaders(sh, obj) {
  // Whitelist of allowed column names per sheet to prevent schema pollution attacks
  const ALLOWED_HEADERS = {
    'Receipts': ['ID','ReceiptNumber','TrackToken','Date','CustomerTitle','CustomerName','CustomerPhone','CustomerEmail','Category','Brand','Model','Accessories','Faults','FaultNotes','Technician','Cost','Deposit','Refunded','PartsCost','PartsUsed','DeliveryDate','Status','Paid','Bonus','CreatedBy','UpdatedBy','UpdatedAt'],
    'Customers': ['Name','Phone','Email','Address','TaxNumber','Debt','Notes'],
    'Technicians': ['Name'],
    'Settings': ['Key','Value'],
    'Payments': ['ID','ReceiptID','Date','Amount','Note','By','PaymentMethod'],
    'Inventory': ['ID','Name','Category','SubCategory','SKU','Barcode','ShelfLocation','Quantity','MinStock','Unit','PurchasePrice','SellPrice','WholesalePrice','WarrantyMonths','CompatibleModels','ImageURL'],
    'Sales': ['ID','Date','ItemsSummary','ItemsJSON','Total','PaymentMethod','AmountPaid','CustomerName','CustomerPhone','By','IsReturned','ReturnDetails'],
    'Returns': ['ID','SaleID','ClientRef','Date','ItemsSummary','RefundAmount','RefundMethod','By'],
    'Quotations': ['ID','Date','ClientName','ClientPhone','ItemsSummary','LaborCost','Total','Status','By'],
    'Services': ['Name','DefaultPrice','Notes'],
    'Purchases': ['ID','Date','Supplier','SupplierID','ItemsSummary','Total','AmountPaid','PaymentMethod','WarrantyMonths','By'],
    'Suppliers': ['ID','Name','Phone','Notes','Debt'],
    'Serials': ['ID','ItemID','ItemName','Serial','PurchaseID','SupplierID','WarrantyMonths','Status','SoldRef','SoldDate'],
    'SupplierPayments': ['ID','SupplierID','PurchaseID','Date','Amount','Note','By'],
    'Expenses': ['ID','Date','Category','Title','Amount','Type','PaymentMethod','Supplier','Reference','AccountCode','Notes','By'],
    'Accounts': ['Code','Name','Type','ParentCode','Nature','Description','Balance'],
    'JournalEntries': ['ID','EntryNumber','Date','Description','ReferenceType','ReferenceID','LinesJSON','TotalDebit','TotalCredit','By'],
    'Invoices': ['ID','InvoiceNumber','Date','DueDate','CustomerName','CustomerPhone','CustomerTaxNumber','CustomerAddress','ItemsSummary','ItemsJSON','Subtotal','TaxPercent','TaxAmount','Discount','Total','AmountPaid','Remaining','Status','PaymentMethod','ReferenceType','ReferenceID','Notes','By'],
    'Users': ['ID','Name','Password','Salt','Role','Sections','Superuser','Notes'],
    'AuditLog': ['Timestamp','User','Action','ReceiptID','RefType','Details','RowHash']
  };
  const lastCol = Math.max(1, sh.getLastColumn());
  let headers = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
  const sheetName = sh.getName();
  const whitelist = ALLOWED_HEADERS[sheetName];
  const allowedKeys = whitelist ? Object.keys(obj).filter(k => k && whitelist.includes(k)) : Object.keys(obj).filter(Boolean);
  const missingKeys = allowedKeys.filter(k => !headers.includes(k));
  if (missingKeys.length > 0) {
    const startCol = headers.length + 1;
    sh.getRange(1, startCol, 1, missingKeys.length).setValues([missingKeys]);
    headers = headers.concat(missingKeys);
  }
  return headers;
}

function sanitizeCellForSheet(val) {
  if (val === null || val === undefined) return '';
  if (typeof val === 'string') {
    const trimmed = val.trim();
    if (trimmed.length > 0 && ('=+-\t\r@'.indexOf(trimmed.charAt(0)) !== -1)) {
      return "'" + val; // Neutralize formula execution in Google Sheets
    }
  }
  return val;
}

/** Input Validation Helpers — Anti-DoS array size limits and NaN prevention */
function validateArraySize(arr, maxSize, label) {
  if (!Array.isArray(arr)) return [];
  if (arr.length > maxSize) {
    throw new Error('تجاوز الحد الأقصى المسموح لعناصر ' + label + ' (' + maxSize + ' عنصر)');
  }
  return arr;
}
function safeNumber(val, defaultVal) {
  const n = Number(val);
  return Number.isFinite(n) ? n : (defaultVal || 0);
}

function appendRow(sheetName, obj) {
  const sh = getSheet(sheetName);
  const headers = ensureHeaders(sh, obj);
  const row = headers.map(h => (obj[h] !== undefined ? sanitizeCellForSheet(obj[h]) : ''));
  sh.appendRow(row);
}

function upsertRow(sheetName, obj, keyField) {
  const sh = getSheet(sheetName);
  const headers = ensureHeaders(sh, obj);
  const keyCol = headers.indexOf(keyField);
  const values = sh.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][keyCol]) === String(obj[keyField])) { rowIndex = i; break; }
  }
  const row = headers.map(h => (obj[h] !== undefined ? sanitizeCellForSheet(obj[h]) : ''));
  if (rowIndex === -1) {
    sh.appendRow(row);
  } else {
    sh.getRange(rowIndex + 1, 1, 1, row.length).setValues([row]);
  }
}

function deleteRow(sheetName, keyValue, keyField) {
  const sh = getSheet(sheetName);
  const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  const keyCol = headers.indexOf(keyField);
  const values = sh.getDataRange().getValues();
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][keyCol]) === String(keyValue)) { sh.deleteRow(i + 1); break; }
  }
}

function logAudit(refId, user, action, refType, detailsJson) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sh = ss.getSheetByName('AuditLog');
    if (!sh) {
      sh = ss.insertSheet('AuditLog');
      sh.appendRow(['Timestamp', 'User', 'Action', 'ReceiptID', 'RefType', 'Details', 'RowHash']);
    } else {
      const lastCol = Math.max(1, sh.getLastColumn());
      const headers = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
      const expectedHeaders = ['Timestamp', 'User', 'Action', 'ReceiptID', 'RefType', 'Details', 'RowHash'];
      const missing = expectedHeaders.filter(h => !headers.includes(h));
      if (missing.length > 0) {
        sh.getRange(1, headers.length + 1, 1, missing.length).setValues([missing]);
      }
    }
    const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(h => String(h || '').trim());
    const hashCol = headers.indexOf('RowHash');
    const lastRow = sh.getLastRow();
    const prev = (lastRow > 1 && hashCol >= 0) ? String(sh.getRange(lastRow, hashCol + 1).getValue() || '') : '';
    const ts = new Date();
    const details = detailsJson ? (typeof detailsJson === 'string' ? detailsJson : JSON.stringify(detailsJson)) : '';
    const rowHash = hashPassword(prev + '|' + ts.toISOString() + '|' + String(user || '') + '|' + String(action || '') + '|' + details);

    const rowObj = {
      Timestamp: ts,
      User: user || '',
      Action: action || '',
      ReceiptID: refId || '',
      RefType: refType || '',
      Details: details,
      RowHash: rowHash
    };
    const rowData = headers.map(h => rowObj[h] !== undefined ? rowObj[h] : '');
    sh.appendRow(rowData);
  } catch(e) {
    console.warn('logAudit failed:', e);
  }
}

function jsonOut(obj, statusCode) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
