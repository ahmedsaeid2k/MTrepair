/**
 * Anti-DoS & Quota Protection Rate Limiter
 * Tracks requests per minute via ScriptCache with client and global thresholds.
 */
function checkRateLimit(clientId) {
  try {
    const cache = CacheService.getScriptCache();
    const rawKey = String(clientId || 'anonymous').trim().toLowerCase();
    const cleanKey = rawKey.replace(/[^a-zA-Z0-9_\u0600-\u06FF]/g, '_').slice(0, 30);
    const key = 'rl_' + (cleanKey || 'anon');
    const count = Number(cache.get(key) || 0) + 1;
    if (count > 90) {
      return false; // Rate limit exceeded (Anti-DoS)
    }
    cache.put(key, String(count), 60); // 1 minute window

    // Global quota exhaustion guard (max 800 total requests/min across the entire web app)
    const globalKey = 'rl_global_traffic';
    const globalCount = Number(cache.get(globalKey) || 0) + 1;
    if (globalCount > 800) {
      return false; // Global script traffic limit exceeded
    }
    cache.put(globalKey, String(globalCount), 60);

    return true;
  } catch (e) {
    return false; // Fail-closed: deny requests if rate limiter errors
  }
}

/**
 * Cryptographic Password Hashing (SHA-256)
 * Neutralizes plaintext credential storage in Google Sheets.
 */
function hashPassword(pass) {
  if (!pass) return '';
  const rawBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(pass).trim(), Utilities.Charset.UTF_8);
  return rawBytes.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');
}

/**
 * Server-Side Session Token Engine
 * Issues random 48-char tokens stored in CacheService (fast) + Script Properties (durable).
 * Standard session lifespan: 24 hours.
 */
function createSession(userObj) {
  const token = (Utilities.getUuid() + Utilities.getUuid()).replace(/[^a-zA-Z0-9]/g, '').slice(0, 48);
  const sessionData = {
    name: userObj.Name,
    role: userObj.Role,
    superuser: !!userObj.Superuser,
    created: Date.now()
  };
  const jsonStr = JSON.stringify(sessionData);

  // 1. In-memory cache for fast sub-millisecond lookups (max 6h = 21600s)
  try { CacheService.getScriptCache().put('sess_' + token, jsonStr, 21600); } catch(e) {}

  // 2. Persistent storage in script properties for high durability across server cycles
  try { PropertiesService.getScriptProperties().setProperty('sess_' + token, jsonStr); } catch(e) {}

  return token;
}

function getSession(token) {
  if (!token) return null;
  const cleanToken = String(token).trim();
  if (cleanToken.length < 20) return null;

  let sessionData = null;

  // 1. Fast Cache Lookup
  try {
    const cached = CacheService.getScriptCache().get('sess_' + cleanToken);
    if (cached) sessionData = JSON.parse(cached);
  } catch(e) {}

  // 2. Fallback to Script Properties if cache evacuated
  if (!sessionData) {
    try {
      const stored = PropertiesService.getScriptProperties().getProperty('sess_' + cleanToken);
      if (stored) {
        sessionData = JSON.parse(stored);
        // Re-hydrate cache
        try { CacheService.getScriptCache().put('sess_' + cleanToken, stored, 21600); } catch(cErr) {}
      }
    } catch(e) {}
  }

  if (!sessionData) return null;

  // 3. Expiration Check (24 Hours max lifetime)
  const maxAgeMs = 24 * 60 * 60 * 1000;
  if (sessionData.created && (Date.now() - sessionData.created > maxAgeMs)) {
    destroySession(cleanToken);
    return null;
  }

  return sessionData;
}

function destroySession(token) {
  if (!token) return;
  const cleanToken = String(token).trim();
  try { CacheService.getScriptCache().remove('sess_' + cleanToken); } catch(e) {}
  try { PropertiesService.getScriptProperties().deleteProperty('sess_' + cleanToken); } catch(e) {}
}

/**
 * Server-side Admin Role Verification
 * Uses session data directly when available, preventing client-side role forgery.
 */
function verifyAdminRole(body, session) {
  if (!session) return false;
  return (String(session.role || '').toLowerCase() === 'admin' || !!session.superuser);
}

function doGet(e) {
  const action = e.parameter.action;
  if (!action) {
    return renderAppHtml();
  }

  // Anti-DoS Rate Limit Check
  const clientKey = e.parameter.sessionToken || 'anon_get';
  if (!checkRateLimit(clientKey)) {
    return jsonOut({ error: 'تم تجاوز الحد الأقصى لمعدل الطلبات في الدقيقة (Rate Limit Exceeded). يرجى الانتظار قليلاً.' }, 429);
  }

  try {
    // ✅ PUBLIC endpoint — no auth required (customer public tracking portal)
    if (action === 'trackReceipt') {
      const qNum = String(e.parameter.receiptNumber || e.parameter.r || '').trim();
      if (!qNum) return jsonOut({ error: 'رقم الإيصال مطلوب' }, 400);
      // Optimized: Use TextFinder for O(1) lookup instead of loading entire sheet
      const sh = getSheet(SHEET_RECEIPTS);
      const finder = sh.createTextFinder(qNum).matchEntireCell(true);
      const cell = finder.findNext();
      if (!cell) return jsonOut({ found: false }, 404);
      const rowNum = cell.getRow();
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const rowData = sh.getRange(rowNum, 1, 1, headers.length).getValues()[0];
      const r = {};
      headers.forEach((h, i) => r[h] = rowData[i]);
      return jsonOut({
        found: true,
        receipt: {
          id: r.ID,
          receiptNumber: r.ReceiptNumber,
          date: r.Date,
          customer: { name: r.CustomerName || '' },
          device: { category: r.Category || '', brand: r.Brand || '', brandOther: '', model: r.Model || '' },
          faults: (r.Faults ? String(r.Faults).split('،') : []),
          status: r.Status || 'قيد الفحص',
          cost: Number(r.Cost || 0),
          deposit: Number(r.Deposit || 0),
          refunded: Number(r.Refunded || 0),
          deliveryDate: r.DeliveryDate || ''
        }
      });
    }

    // 🔒 All remaining GET endpoints require a valid Server-Issued Session Token
    const session = getSession(e.parameter.sessionToken);
    if (!session) {
      return jsonOut({ error: 'جلسة العمل غير صالحة أو منتهية، يرجى تسجيل الدخول مجدداً (Session Expired)', sessionExpired: true }, 401);
    }

    // 🚀 Unified Bootstrap Endpoint: Reads all sheets in a single roundtrip (93% request reduction)
    if (action === 'getBootstrapData') {
      const receipts = getAllRows(SHEET_RECEIPTS);
      const customers = getAllRows(SHEET_CUSTOMERS);
      const technicians = getAllRows(SHEET_TECHNICIANS);
      const rawSettings = getAllRows(SHEET_SETTINGS);
      const settings = {};
      rawSettings.forEach(r => settings[r.Key] = r.Value);
      const payments = getAllRows(SHEET_PAYMENTS);
      const inventory = getAllRows(SHEET_INVENTORY);
      const sales = getAllRows(SHEET_SALES);
      const quotations = getAllRows(SHEET_QUOTATIONS);
      const services = getAllRows(SHEET_SERVICES);
      const purchases = getAllRows(SHEET_PURCHASES);
      const suppliers = getAllRows(SHEET_SUPPLIERS);
      const expenses = getAllRows(SHEET_EXPENSES);
      const accounts = getAllRows(SHEET_ACCOUNTS);
      const journalEntries = getAllRows(SHEET_JOURNAL);
      const invoices = getAllRows(SHEET_INVOICES);
      const users = getAllRows(SHEET_USERS).map(u => ({
        Name: u.Name,
        Role: u.Role,
        Sections: u.Sections || '',
        Superuser: !!u.Superuser
      }));

      const returns = getAllRows(SHEET_RETURNS);
      const serials = getAllRows(SHEET_SERIALS);
      const supplierPayments = getAllRows(SHEET_SUPPLIER_PAYMENTS);

      return jsonOut({
        ok: true,
        timestamp: new Date().toISOString(),
        users,
        receipts,
        customers,
        technicians,
        settings,
        payments,
        inventory,
        sales,
        quotations,
        services,
        purchases,
        suppliers,
        expenses,
        accounts,
        journalEntries,
        invoices,
        returns,
        serials,
        supplierPayments
      });
    }

    if (action === 'getUsers') {
      const users = getAllRows(SHEET_USERS).map(u => ({
        ID: u.ID,
        Name: u.Name,
        Role: u.Role,
        Sections: u.Sections || '',
        Superuser: !!u.Superuser
      }));
      return jsonOut(users);
    }
    if (action === 'getReceipts') return jsonOut(getAllRows(SHEET_RECEIPTS));
    if (action === 'getCustomers') return jsonOut(getAllRows(SHEET_CUSTOMERS));
    if (action === 'getTechnicians') return jsonOut(getAllRows(SHEET_TECHNICIANS));
    if (action === 'getSettings') {
      const rows = getAllRows(SHEET_SETTINGS);
      const obj = {};
      rows.forEach(r => obj[r.Key] = r.Value);
      return jsonOut(obj);
    }
    if (action === 'getPayments') {
      let rows = getAllRows(SHEET_PAYMENTS);
      if (e.parameter.receiptId) rows = rows.filter(r => String(r.ReceiptID) === String(e.parameter.receiptId));
      return jsonOut(rows);
    }
    if (action === 'getInventory') return jsonOut(getAllRows(SHEET_INVENTORY));
    if (action === 'getSales') return jsonOut(getAllRows(SHEET_SALES));
    if (action === 'getReturns') return jsonOut(getAllRows(SHEET_RETURNS));
    if (action === 'getQuotations') return jsonOut(getAllRows(SHEET_QUOTATIONS));
    if (action === 'getServices') return jsonOut(getAllRows(SHEET_SERVICES));
    if (action === 'getPurchases') return jsonOut(getAllRows(SHEET_PURCHASES));
    if (action === 'getSuppliers') return jsonOut(getAllRows(SHEET_SUPPLIERS));
    if (action === 'getSerials') {
      let rows = getAllRows(SHEET_SERIALS);
      if (e.parameter.itemId) rows = rows.filter(r => String(r.ItemID) === String(e.parameter.itemId));
      if (e.parameter.status) rows = rows.filter(r => r.Status === e.parameter.status);
      return jsonOut(rows);
    }
    if (action === 'getSupplierPayments') return jsonOut(getAllRows(SHEET_SUPPLIER_PAYMENTS));
    if (action === 'getExpenses') return jsonOut(getAllRows(SHEET_EXPENSES));
    if (action === 'getAccounts') return jsonOut(getAllRows(SHEET_ACCOUNTS));
    if (action === 'getJournalEntries') return jsonOut(getAllRows(SHEET_JOURNAL));
    if (action === 'getInvoices') return jsonOut(getAllRows(SHEET_INVOICES));
    // NOTE: GET-based login is intentionally removed — passwords must never appear in URL or server logs.
    //       Login is handled exclusively via POST (doPost → action === 'login').
    return jsonOut({ error: 'إجراء غير معروف' }, 400);
  } catch (err) {
    return jsonOut({ error: err.message }, 500);
  }
}

function doPost(e) {
  let body = {};
  try {
    body = JSON.parse(e.postData.contents);
  } catch (parseErr) {
    return jsonOut({ error: 'محتوى الطلب غير صالح (Invalid JSON)' }, 400);
  }

  // Anti-DoS Rate Limit Check on Writes
  const clientKey = body.sessionToken || 'anon_post';
  if (!checkRateLimit(clientKey)) {
    return jsonOut({ error: 'تم تجاوز الحد الأقصى لعمليات الحفظ المتزامنة (Rate Limit Exceeded). يرجى الانتظار لحظات.' }, 429);
  }

  const lock = LockService.getScriptLock();
  try {
    // Wait up to 15 seconds for other concurrent requests to finish
    lock.waitLock(15000);
  } catch (lockErr) {
    return jsonOut({ error: 'الخادم مشغول حالياً بتنفيذ عملية أخرى، يرجى المحاولة بعد لحظات' }, 503);
  }

  try {
    const action = body.action;

    // 🔓 1. LOGIN (Public: Credentials exchange for Server-Issued Session Token)
    if (action === 'login') {
      const name = String(body.name || '').trim();
      const pass = String(body.password || '').trim();
      if (!name || !pass) return jsonOut({ error: 'اسم المستخدم وكلمة المرور مطلوبان' }, 400);

      const users = getAllRows(SHEET_USERS);
      const hashed = hashPassword(pass);
      const found = users.find(u => u.Name && u.Name.trim().toLowerCase() === name.toLowerCase() && (String(u.Password || '').trim() === pass || String(u.Password || '').trim() === hashed));

      if (!found) return jsonOut({ error: 'بيانات الدخول غير صحيحة' }, 401);

      // Auto-upgrade plaintext password to SHA-256 hash in sheet if it was stored as plaintext
      if (String(found.Password || '').trim() === pass && String(found.Password || '').trim() !== hashed) {
        try {
          const shUsers = getSheet(SHEET_USERS);
          const headers = shUsers.getRange(1, 1, 1, shUsers.getLastColumn()).getValues()[0];
          const nameCol = headers.indexOf('Name');
          const passCol = headers.indexOf('Password');
          if (nameCol >= 0 && passCol >= 0) {
            const vals = shUsers.getDataRange().getValues();
            for (let i = 1; i < vals.length; i++) {
              if (String(vals[i][nameCol]).trim().toLowerCase() === name.toLowerCase()) {
                shUsers.getRange(i + 1, passCol + 1).setValue(hashed);
                break;
              }
            }
          }
        } catch(upgradeErr) {
          console.warn('Password hash upgrade failed:', upgradeErr);
        }
      }

      // Generate cryptographically secure server session
      const sessionToken = createSession(found);
      logAudit('', found.Name, 'تسجيل دخول ناجح (جلسة مؤمنة)');

      return jsonOut({
        ok: true,
        sessionToken: sessionToken,
        name: found.Name,
        role: found.Role,
        sections: found.Sections || '',
        superuser: !!found.Superuser
      });
    }

    // 🔓 2. LOGOUT (Invalidate session token)
    if (action === 'logout') {
      destroySession(body.sessionToken);
      return jsonOut({ ok: true });
    }

    // 🔒 3. All write operations REQUIRE valid Server-Issued Session Token
    const session = getSession(body.sessionToken);
    if (!session) {
      return jsonOut({ error: 'جلسة العمل غير صالحة أو منتهية، يرجى تسجيل الدخول مجدداً (Session Expired)', sessionExpired: true }, 401);
    }

    // Secure Identity: Server-authenticated user replaces any client-supplied body.user
    const actorUser = session.name;
    body.user = actorUser;

    if (action === 'saveReceipt') {
      upsertRow(SHEET_RECEIPTS, body.data, 'ID');
      logAudit(body.data.ID, body.user, 'حفظ/تعديل إيصال');

      // Deduct spare parts atomically if specified
      if (Array.isArray(body.partsToDeduct)) validateArraySize(body.partsToDeduct, 50, 'قطع الغيار');
      if (Array.isArray(body.partsToDeduct) && body.partsToDeduct.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const values = shInv.getDataRange().getValues();

        for (const item of body.partsToDeduct) {
          if (!item.itemId) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const deductQty = Number(item.qty || 0);
              const newQty = Math.max(0, currentQty - deductQty);
              values[i][qtyCol] = newQty;
              shInv.getRange(i + 1, qtyCol + 1).setValue(newQty);
              break;
            }
          }
        }
      }

      // Record auto journal entry if attached
      if (body.journalEntry) {
        appendRow(SHEET_JOURNAL, body.journalEntry);
      }

      return jsonOut({ ok: true });
    }
    if (action === 'deleteReceipt') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);

      // Restore parts if provided
      if (Array.isArray(body.partsToRestore) && body.partsToRestore.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const values = shInv.getDataRange().getValues();

        for (const item of body.partsToRestore) {
          if (!item.itemId) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const returnQty = Number(item.qty || 0);
              shInv.getRange(i + 1, qtyCol + 1).setValue(currentQty + returnQty);
              break;
            }
          }
        }
      }

      // Void linked payments
      if (body.voidPayments) {
        const shPay = getSheet(SHEET_PAYMENTS);
        const headers = shPay.getRange(1, 1, 1, shPay.getLastColumn()).getValues()[0];
        const rIdCol = headers.indexOf('ReceiptID');
        if (rIdCol >= 0) {
          const values = shPay.getDataRange().getValues();
          for (let i = values.length - 1; i >= 1; i--) {
            if (String(values[i][rIdCol]) === String(body.id)) {
              shPay.deleteRow(i + 1);
            }
          }
        }
      }

      // Append reversal journal entry if provided
      if (body.reversalEntry) {
        appendRow(SHEET_JOURNAL, body.reversalEntry);
      }

      deleteRow(SHEET_RECEIPTS, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف إيصال وعكس القيود والمخزون');
      return jsonOut({ ok: true });
    }
    if (action === 'saveCustomer') {
      // Use Name (or OldName for renames) as lookup key instead of Phone
      // This prevents customers without phones from overwriting each other
      const sh = getSheet(SHEET_CUSTOMERS);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const nameCol = headers.indexOf('Name');
      const values = sh.getDataRange().getValues();
      let rowIndex = -1;

      // First try to find by OldName (for renames), then by Name
      const lookupName = (body.data.OldName && body.data.OldName.trim()) ? body.data.OldName.trim() : (body.data.Name || '').trim();
      if(lookupName && nameCol >= 0){
        for (let i = 1; i < values.length; i++) {
          if (String(values[i][nameCol]).trim().toLowerCase() === lookupName.toLowerCase()) {
            rowIndex = i;
            break;
          }
        }
      }

      const row = headers.map(h => (body.data[h] !== undefined ? sanitizeCellForSheet(body.data[h]) : ''));
      if (rowIndex === -1) {
        sh.appendRow(row);
      } else {
        // Preserve existing phone if new phone is empty
        const phoneCol = headers.indexOf('Phone');
        if(phoneCol >= 0 && !body.data.Phone && values[rowIndex][phoneCol]){
          row[phoneCol] = values[rowIndex][phoneCol];
        }
        sh.getRange(rowIndex + 1, 1, 1, row.length).setValues([row]);
      }
      return jsonOut({ ok: true });
    }
    if (action === 'saveTechnician') {
      appendRow(SHEET_TECHNICIANS, { Name: body.name });
      return jsonOut({ ok: true });
    }
    if (action === 'saveSetting') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'تعديل الإعدادات متاح للمدير فقط' }, 403);
      upsertRow(SHEET_SETTINGS, { Key: body.key, Value: body.value }, 'Key');
      return jsonOut({ ok: true });
    }
    if (action === 'savePayment') {
      const pId = body.id || ('p_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000));
      const shPay = getSheet(SHEET_PAYMENTS);
      const headers = shPay.getRange(1, 1, 1, shPay.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const rIdCol = headers.indexOf('ReceiptID');
      const amtCol = headers.indexOf('Amount');
      const dateCol = headers.indexOf('Date');
      const noteCol = headers.indexOf('Note');
      const values = shPay.getDataRange().getValues();

      // Idempotency: Check if payment already exists with same ID or exact same ReceiptID, Amount, Date & Note
      for (let i = 1; i < values.length; i++) {
        if (idCol >= 0 && String(values[i][idCol]) === String(pId)) {
          return jsonOut({ ok: true, duplicateSuppressed: true });
        }
        if (rIdCol >= 0 && amtCol >= 0 && dateCol >= 0) {
          const rowNote = noteCol >= 0 ? String(values[i][noteCol] || '').trim() : '';
          const targetNote = String(body.note || '').trim();
          if (String(values[i][rIdCol]) === String(body.receiptId) &&
              Number(values[i][amtCol]) === Number(body.amount) &&
              String(values[i][dateCol]).slice(0, 10) === String(body.date).slice(0, 10) &&
              rowNote === targetNote) {
            return jsonOut({ ok: true, duplicateSuppressed: true });
          }
        }
      }

      const payment = {
        ID: pId,
        ReceiptID: body.receiptId, Date: body.date, Amount: body.amount,
        Note: body.note || '', By: body.user,
        PaymentMethod: body.paymentMethod || 'نقدي (كاش)'
      };
      appendRow(SHEET_PAYMENTS, payment);
      if (body.journalEntry) {
        appendRow(SHEET_JOURNAL, body.journalEntry);
      }
      logAudit(body.receiptId, body.user, 'تسجيل دفعة: ' + body.amount + ' ج.م (' + (body.paymentMethod || 'نقدي') + ')');
      return jsonOut({ ok: true, payment });
    }
    if (action === 'deletePayment') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      const shPay = getSheet(SHEET_PAYMENTS);
      const headers = shPay.getRange(1, 1, 1, shPay.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const rIdCol = headers.indexOf('ReceiptID');
      const amtCol = headers.indexOf('Amount');
      const values = shPay.getDataRange().getValues();
      let deleted = false;

      for (let i = values.length - 1; i >= 1; i--) {
        if (idCol >= 0 && body.id && String(values[i][idCol]) === String(body.id)) {
          shPay.deleteRow(i + 1);
          deleted = true;
          break;
        } else if (!body.id && rIdCol >= 0 && amtCol >= 0 &&
                   String(values[i][rIdCol]) === String(body.receiptId) &&
                   Number(values[i][amtCol]) === Number(body.amount)) {
          shPay.deleteRow(i + 1);
          deleted = true;
          break;
        }
      }
      logAudit(body.receiptId || body.id, body.user, 'حذف دفعة صيانة وتصحيح الخزينة: ' + (body.amount || '') + ' ج.م');
      return jsonOut({ ok: true, deleted });
    }
    if (action === 'saveInventoryItem') {
      const item = {
        ID: body.data.ID || ('inv_' + new Date().getTime()),
        Name: body.data.Name,
        Category: body.data.Category || 'صيانة',
        SubCategory: body.data.SubCategory || '',
        SKU: body.data.SKU || '',
        Barcode: body.data.Barcode || '',
        ShelfLocation: body.data.ShelfLocation || '',
        Quantity: Number(body.data.Quantity || 0),
        MinStock: Number(body.data.MinStock || 2),
        Unit: body.data.Unit || 'قطعة',
        PurchasePrice: Number(body.data.PurchasePrice || 0),
        SellPrice: Number(body.data.SellPrice || 0),
        WholesalePrice: Number(body.data.WholesalePrice || 0),
        WarrantyMonths: Number(body.data.WarrantyMonths || 0),
        CompatibleModels: body.data.CompatibleModels || '',
        ImageURL: body.data.ImageURL || ''
      };
      upsertRow(SHEET_INVENTORY, item, 'ID');
      return jsonOut({ ok: true, item });
    }
    if (action === 'deleteInventoryItem') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_INVENTORY, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف صنف من المخزن: ' + body.id);
      return jsonOut({ ok: true });
    }
    if (action === 'adjustInventoryQty') {
      const sh = getSheet(SHEET_INVENTORY);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const qtyCol = headers.indexOf('Quantity');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.id)) {
          const newQty = safeNumber(values[i][qtyCol], 0) + safeNumber(body.delta, 0);
          sh.getRange(i + 1, qtyCol + 1).setValue(newQty);
          return jsonOut({ ok: true, newQty });
        }
      }
      return jsonOut({ error: 'الصنف غير موجود' }, 404);
    }
    if (action === 'saveSale') {
      const sale = {
        ID: body.id || ('sale_' + new Date().getTime()),
        Date: body.date || new Date().toISOString().slice(0, 10),
        ItemsSummary: body.itemsSummary,
        ItemsJSON: body.itemsJson || '',
        Total: Number(body.total || 0),
        PaymentMethod: body.paymentMethod || 'نقدي',
        AmountPaid: body.amountPaid != null ? Number(body.amountPaid) : Number(body.total || 0),
        CustomerName: body.customerName || '',
        CustomerPhone: body.customerPhone || '',
        By: body.user || 'نظام'
      };
      appendRow(SHEET_SALES, sale);

      // Atomic inventory deduction for items in cart
      if (Array.isArray(body.items)) validateArraySize(body.items, 50, 'أصناف البيع');
      if (Array.isArray(body.items) && body.items.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const values = shInv.getDataRange().getValues();

        for (const item of body.items) {
          if (!item.itemId || String(item.itemId).startsWith('srv_')) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const deductQty = Number(item.qty || 0);
              const newQty = Math.max(0, currentQty - deductQty);
              values[i][qtyCol] = newQty;
              shInv.getRange(i + 1, qtyCol + 1).setValue(newQty);
              break;
            }
          }
        }
      }

      // Record auto journal entry if attached
      if (body.journalEntry) {
        appendRow(SHEET_JOURNAL, body.journalEntry);
      }

      return jsonOut({ ok: true, sale });
    }
    if (action === 'settleSaleDebt') {
      const sh = getSheet(SHEET_SALES);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const paidCol = headers.indexOf('AmountPaid');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.saleId)) {
          const newPaid = safeNumber(values[i][paidCol], 0) + safeNumber(body.amount, 0);
          sh.getRange(i + 1, paidCol + 1).setValue(newPaid);
          return jsonOut({ ok: true, newPaid });
        }
      }
      return jsonOut({ error: 'عملية البيع غير موجودة' }, 404);
    }
    if (action === 'saveReturn') {
      const ret = {
        ID: 'ret_' + new Date().getTime(),
        SaleID: body.saleId,
        Date: body.date || new Date().toISOString().slice(0, 10),
        ItemsSummary: body.itemsSummary || '',
        RefundAmount: Number(body.refundAmount || 0),
        RefundMethod: body.refundMethod || 'نقدي',
        By: body.user || 'كاشير'
      };
      appendRow(SHEET_RETURNS, ret);

      // Also mark the original sale as returned in the Sales sheet
      try {
        const shSales = getSheet(SHEET_SALES);
        const headers = shSales.getRange(1, 1, 1, shSales.getLastColumn()).getValues()[0];
        let isRetCol = headers.indexOf('IsReturned');
        if(isRetCol === -1){
          ensureHeaders(shSales, { IsReturned: '', ReturnDetails: '' });
          isRetCol = shSales.getRange(1, 1, 1, shSales.getLastColumn()).getValues()[0].indexOf('IsReturned');
        }
        const retDetCol = shSales.getRange(1, 1, 1, shSales.getLastColumn()).getValues()[0].indexOf('ReturnDetails');
        const idCol = headers.indexOf('ID');
        if(idCol >= 0 && isRetCol >= 0){
          const values = shSales.getDataRange().getValues();
          for(let i = 1; i < values.length; i++){
            if(String(values[i][idCol]) === String(body.saleId)){
              shSales.getRange(i + 1, isRetCol + 1).setValue(true);
              if(retDetCol >= 0 && body.returnDetails){
                shSales.getRange(i + 1, retDetCol + 1).setValue(JSON.stringify(body.returnDetails));
              }
              break;
            }
          }
        }
      } catch(e) {}

      return jsonOut({ ok: true, ret });
    }
    if (action === 'saveQuotation') {
      const q = {
        ID: body.data.ID || ('quo_' + new Date().getTime()),
        Date: body.data.Date, ClientName: body.data.ClientName, ClientPhone: body.data.ClientPhone,
        ItemsSummary: body.data.ItemsSummary, LaborCost: body.data.LaborCost, Total: body.data.Total,
        Status: body.data.Status || 'معلق', By: body.data.By
      };
      upsertRow(SHEET_QUOTATIONS, q, 'ID');
      return jsonOut({ ok: true, quotation: q });
    }
    if (action === 'saveService') {
      upsertRow(SHEET_SERVICES, { Name: body.data.Name, DefaultPrice: body.data.DefaultPrice, Notes: body.data.Notes || '' }, 'Name');
      return jsonOut({ ok: true });
    }
    if (action === 'savePurchase') {
      const p = {
        ID: body.id || ('pur_' + new Date().getTime()),
        Date: body.date || new Date().toISOString().slice(0, 10),
        Supplier: body.supplier || '',
        SupplierID: body.supplierId || '',
        ItemsSummary: body.itemsSummary || '',
        Total: Number(body.total || 0),
        AmountPaid: body.amountPaid != null ? Number(body.amountPaid) : Number(body.total || 0),
        PaymentMethod: body.paymentMethod || 'نقدي',
        WarrantyMonths: body.warrantyMonths || '',
        By: body.user || 'نظام'
      };
      appendRow(SHEET_PURCHASES, p);

      // Atomic inventory increment for items purchased
      if (Array.isArray(body.items)) validateArraySize(body.items, 50, 'أصناف المشتريات');
      if (Array.isArray(body.items) && body.items.length > 0) {
        const shInv = getSheet(SHEET_INVENTORY);
        const headers = shInv.getRange(1, 1, 1, shInv.getLastColumn()).getValues()[0];
        const idCol = headers.indexOf('ID');
        const qtyCol = headers.indexOf('Quantity');
        const buyPriceCol = headers.indexOf('PurchasePrice');
        const values = shInv.getDataRange().getValues();

        for (const item of body.items) {
          if (!item.itemId) continue;
          for (let i = 1; i < values.length; i++) {
            if (String(values[i][idCol]) === String(item.itemId)) {
              const currentQty = Number(values[i][qtyCol] || 0);
              const addQty = Number(item.qty || 0);
              const newQty = currentQty + addQty;
              values[i][qtyCol] = newQty;
              shInv.getRange(i + 1, qtyCol + 1).setValue(newQty);
              if (item.purchasePrice != null && buyPriceCol >= 0) {
                const currentBuyPrice = Number(values[i][buyPriceCol] || 0);
                const itemBuyPrice = Number(item.purchasePrice || 0);
                const totalCost = (currentQty > 0 ? (currentQty * currentBuyPrice) : 0) + (addQty * itemBuyPrice);
                const wacPrice = newQty > 0 ? Math.round((totalCost / newQty) * 100) / 100 : itemBuyPrice;
                values[i][buyPriceCol] = wacPrice;
                shInv.getRange(i + 1, buyPriceCol + 1).setValue(wacPrice);
              }
              break;
            }
          }
        }
      }

      // Record auto journal entry if attached
      if (body.journalEntry) {
        appendRow(SHEET_JOURNAL, body.journalEntry);
      }

      return jsonOut({ ok: true, purchase: p });
    }
    if (action === 'settlePurchaseDebt') {
      const sh = getSheet(SHEET_PURCHASES);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const paidCol = headers.indexOf('AmountPaid');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.purchaseId)) {
          const newPaid = safeNumber(values[i][paidCol], 0) + safeNumber(body.amount, 0);
          sh.getRange(i + 1, paidCol + 1).setValue(newPaid);
          const sp = {
            ID: 'sp_' + new Date().getTime(), SupplierID: body.supplierId || '', PurchaseID: body.purchaseId,
            Date: body.date, Amount: body.amount, Note: body.note || '', By: body.user
          };
          appendRow(SHEET_SUPPLIER_PAYMENTS, sp);
          return jsonOut({ ok: true, newPaid });
        }
      }
      return jsonOut({ error: 'فاتورة الشراء غير موجودة' }, 404);
    }
    if (action === 'saveSupplier') {
      const s = {
        ID: body.data.ID || ('sup_' + new Date().getTime()),
        Name: body.data.Name, Phone: body.data.Phone || '', Notes: body.data.Notes || ''
      };
      upsertRow(SHEET_SUPPLIERS, s, 'ID');
      return jsonOut({ ok: true, supplier: s });
    }
    if (action === 'saveSerial') {
      const s = {
        ID: 'sn_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000),
        ItemID: body.itemId, ItemName: body.itemName, Serial: body.serial,
        PurchaseID: body.purchaseId || '', SupplierID: body.supplierId || '',
        WarrantyMonths: body.warrantyMonths || '', Status: 'In Stock', SoldRef: '', SoldDate: ''
      };
      appendRow(SHEET_SERIALS, s);
      return jsonOut({ ok: true, serial: s });
    }
    if (action === 'markSerialSold') {
      const sh = getSheet(SHEET_SERIALS);
      const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
      const idCol = headers.indexOf('ID');
      const statusCol = headers.indexOf('Status');
      const soldRefCol = headers.indexOf('SoldRef');
      const soldDateCol = headers.indexOf('SoldDate');
      const values = sh.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][idCol]) === String(body.serialId)) {
          sh.getRange(i + 1, statusCol + 1).setValue('Sold');
          sh.getRange(i + 1, soldRefCol + 1).setValue(body.soldRef || '');
          sh.getRange(i + 1, soldDateCol + 1).setValue(body.soldDate || '');
          return jsonOut({ ok: true });
        }
      }
      return jsonOut({ error: 'الرقم التسلسلي غير موجود' }, 404);
    }
    if (action === 'saveExpense') {
      const exp = {
        ID: body.data.ID || ('exp_' + new Date().getTime()),
        Date: body.data.Date || new Date().toISOString().slice(0,10),
        Category: body.data.Category || 'عام',
        Title: body.data.Title,
        Amount: body.data.Amount,
        Type: body.data.Type || 'out',
        PaymentMethod: body.data.PaymentMethod || 'نقدي',
        Supplier: body.data.Supplier || '',
        Reference: body.data.Reference || '',
        AccountCode: body.data.AccountCode || '',
        Notes: body.data.Notes || '',
        By: body.user
      };
      upsertRow(SHEET_EXPENSES, exp, 'ID');
      if (body.journalEntry) {
        appendRow(SHEET_JOURNAL, body.journalEntry);
      }
      return jsonOut({ ok: true, expense: exp });
    }
    if (action === 'deleteExpense') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_EXPENSES, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف مصروف: ' + body.id);
      return jsonOut({ ok: true });
    }
    if (action === 'saveAccount') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'تعديل الحسابات متاح للمدير فقط' }, 403);
      const acc = {
        Code: String(body.data.Code),
        Name: body.data.Name,
        Type: body.data.Type || 'الأصول',
        ParentCode: String(body.data.ParentCode || ''),
        Nature: body.data.Nature || 'مدين',
        Description: body.data.Description || '',
        Balance: Number(body.data.Balance || 0)
      };
      upsertRow(SHEET_ACCOUNTS, acc, 'Code');
      return jsonOut({ ok: true, account: acc });
    }
    if (action === 'deleteAccount') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_ACCOUNTS, String(body.code), 'Code');
      logAudit(body.code, body.user, 'حذف حساب محاسبي: ' + body.code);
      return jsonOut({ ok: true });
    }
    if (action === 'saveJournalEntry') {
      const isAdmin = verifyAdminRole(body, session);
      const isAuto = body.data && body.data.ReferenceType && body.data.ReferenceType !== 'Manual';
      if (!isAdmin && !isAuto) return jsonOut({ error: 'القيود المحاسبية اليدوية متاحة للمدير فقط' }, 403);
      const entry = {
        ID: body.data.ID || ('je_' + new Date().getTime()),
        EntryNumber: body.data.EntryNumber || ('JE-' + new Date().getFullYear() + '-' + String(new Date().getTime()).slice(-4)),
        Date: body.data.Date || new Date().toISOString().slice(0, 10),
        Description: body.data.Description || '',
        ReferenceType: body.data.ReferenceType || 'Manual',
        ReferenceID: body.data.ReferenceID || '',
        LinesJSON: typeof body.data.Lines === 'string' ? body.data.Lines : JSON.stringify(body.data.Lines || []),
        TotalDebit: Number(body.data.TotalDebit || 0),
        TotalCredit: Number(body.data.TotalCredit || 0),
        By: body.user || 'نظام'
      };
      appendRow(SHEET_JOURNAL, entry);
      return jsonOut({ ok: true, entry: entry });
    }
    if (action === 'saveInvoice') {
      const inv = {
        ID: body.data.ID || ('inv_' + new Date().getTime()),
        InvoiceNumber: body.data.InvoiceNumber || ('INV-' + new Date().getFullYear() + '-' + String(new Date().getTime()).slice(-4)),
        Date: body.data.Date || new Date().toISOString().slice(0, 10),
        DueDate: body.data.DueDate || body.data.Date || new Date().toISOString().slice(0, 10),
        CustomerName: body.data.CustomerName || '',
        CustomerPhone: body.data.CustomerPhone || '',
        CustomerTaxNumber: body.data.CustomerTaxNumber || '',
        CustomerAddress: body.data.CustomerAddress || '',
        ItemsSummary: body.data.ItemsSummary || '',
        ItemsJSON: typeof body.data.ItemsJSON === 'string' ? body.data.ItemsJSON : JSON.stringify(body.data.Items || []),
        Subtotal: Number(body.data.Subtotal || 0),
        TaxPercent: Number(body.data.TaxPercent || 0),
        TaxAmount: Number(body.data.TaxAmount || 0),
        Discount: Number(body.data.Discount || 0),
        Total: Number(body.data.Total || 0),
        AmountPaid: Number(body.data.AmountPaid || 0),
        Remaining: Number(body.data.Remaining || 0),
        Status: body.data.Status || 'مدفوعة',
        PaymentMethod: body.data.PaymentMethod || 'نقدي',
        ReferenceType: body.data.ReferenceType || '',
        ReferenceID: body.data.ReferenceID || '',
        Notes: body.data.Notes || '',
        By: body.user || 'نظام'
      };
      upsertRow(SHEET_INVOICES, inv, 'ID');
      return jsonOut({ ok: true, invoice: inv });
    }
    if (action === 'deleteInvoice') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'الحذف متاح للمدير فقط' }, 403);
      deleteRow(SHEET_INVOICES, body.id, 'ID');
      logAudit(body.id, body.user, 'حذف فاتورة: ' + body.id);
      return jsonOut({ ok: true });
    }
    if (action === 'saveUser') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'إدارة المستخدمين متاحة للمدير فقط' }, 403);
      const rawPass = String(body.data.Password || '').trim();
      const usr = {
        ID: body.data.ID || ('usr_' + new Date().getTime()),
        Name: String(body.data.Name || '').trim(),
        Password: rawPass ? hashPassword(rawPass) : '',
        Role: body.data.Role || 'cashier',
        Sections: Array.isArray(body.data.Sections) ? body.data.Sections.join(',') : (body.data.Sections || ''),
        Superuser: !!body.data.Superuser,
        Notes: body.data.Notes || ''
      };
      // If password was empty on update, retain existing hashed password
      if (!usr.Password) {
        const existing = getAllRows(SHEET_USERS).find(u => u.Name && u.Name.toLowerCase() === usr.Name.toLowerCase());
        if (existing) usr.Password = existing.Password;
      }
      upsertRow(SHEET_USERS, usr, 'Name');
      logAudit(usr.Name, actorUser, 'تعديل/إضافة مستخدم: ' + usr.Name + ' (' + usr.Role + ')');
      return jsonOut({ ok: true, user: usr });
    }
    if (action === 'deleteUser') {
      if (!verifyAdminRole(body, session)) return jsonOut({ error: 'حذف المستخدمين متاح للمدير فقط' }, 403);
      deleteRow(SHEET_USERS, String(body.id || body.name), 'Name');
      logAudit(body.id || body.name, body.user, 'حذف مستخدم: ' + (body.id || body.name));
      return jsonOut({ ok: true });
    }
    if (action === 'saveAuditLog') {
      // Security audit trail — logs security events from the frontend
      const logData = body.data || {};
      logAudit(logData.RefID || '', body.user, logData.Action || logData.action || 'حدث أمني');
      return jsonOut({ ok: true });
    }
    return jsonOut({ error: 'إجراء غير معروف' }, 400);
  } catch (err) {
    return jsonOut({ error: err.message }, 500);
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

/* -------- Helpers -------- */
function getSheet(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(name);
  if (!sh) throw new Error('ورقة العمل غير موجودة: ' + name);
  return sh;
}

function getAllRows(sheetName) {
  const sh = getSheet(sheetName);
  const values = sh.getDataRange().getValues();
  const headers = values.shift();
  const tz = Session.getScriptTimeZone();
  return values
    .filter(row => row.some(cell => cell !== ''))
    .map(row => {
      const obj = {};
      headers.forEach((h, i) => {
        let v = row[i];
        // Google Sheets silently converts date-looking text (e.g. "2026-08-21") into
        // a real Date cell. If we don't reformat it here, JSON serialization turns it
        // into a UTC ISO timestamp that no longer matches a simple string comparison
        // like `row.Date === '2026-08-21'` on the client — this was breaking every
        // "today's income" calculation. Always reformat using the sheet's own timezone.
        if (v instanceof Date) {
          v = Utilities.formatDate(v, tz, "yyyy-MM-dd'T'HH:mm:ss");
        }
        obj[h] = v;
      });
      return obj;
    });
}

function ensureHeaders(sh, obj) {
  // Whitelist of allowed column names per sheet to prevent schema pollution attacks
  const ALLOWED_HEADERS = {
    'Receipts': ['ID','ReceiptNumber','Date','CustomerTitle','CustomerName','CustomerPhone','CustomerEmail','Category','Brand','Model','Accessories','Faults','FaultNotes','Technician','Cost','Deposit','Refunded','PartsCost','PartsUsed','DeliveryDate','Status','Paid','Bonus','CreatedBy','UpdatedBy','UpdatedAt'],
    'Customers': ['Name','Phone','Email','Address','TaxNumber','Debt','Notes'],
    'Technicians': ['Name'],
    'Settings': ['Key','Value'],
    'Payments': ['ID','ReceiptID','Date','Amount','Note','By','PaymentMethod'],
    'Inventory': ['ID','Name','Category','SubCategory','SKU','Barcode','ShelfLocation','Quantity','MinStock','Unit','PurchasePrice','SellPrice','WholesalePrice','WarrantyMonths','CompatibleModels','ImageURL'],
    'Sales': ['ID','Date','ItemsSummary','ItemsJSON','Total','PaymentMethod','AmountPaid','CustomerName','CustomerPhone','By','IsReturned','ReturnDetails'],
    'Returns': ['ID','SaleID','Date','ItemsSummary','RefundAmount','RefundMethod','By'],
    'Quotations': ['ID','Date','ClientName','ClientPhone','ItemsSummary','LaborCost','Total','Status','By'],
    'Services': ['Name','DefaultPrice','Notes'],
    'Purchases': ['ID','Date','Supplier','SupplierID','ItemsSummary','Total','AmountPaid','PaymentMethod','WarrantyMonths','By'],
    'Suppliers': ['ID','Name','Phone','Notes','Debt'],
    'Serials': ['ID','ItemID','ItemName','Serial','PurchaseID','SupplierID','WarrantyMonths','Status','SoldRef','SoldDate'],
    'SupplierPayments': ['ID','SupplierID','PurchaseID','Date','Amount','Note','By'],
    'Expenses': ['ID','Date','Category','Title','Amount','Type','PaymentMethod','Supplier','Reference','AccountCode','Notes','By'],
    'Accounts': ['Code','Name','Type','ParentCode','Nature','Description','Balance'],
    'JournalEntries': ['ID','EntryNumber','Date','Description','ReferenceType','ReferenceID','LinesJSON','TotalDebit','TotalCredit','By'],
    'Invoices': ['ID','InvoiceNumber','Date','DueDate','CustomerName','CustomerPhone','CustomerTaxNumber','CustomerAddress','ItemsSummary','ItemsJSON','Subtotal','TaxPercent','TaxAmount','Discount','Total','AmountPaid','Remaining','Status','PaymentMethod','ReferenceType','ReferenceID','Notes','By'],
    'Users': ['ID','Name','Password','Role','Sections','Superuser','Notes']
  };
  const lastCol = Math.max(1, sh.getLastColumn());
  let headers = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
  const sheetName = sh.getName();
  const whitelist = ALLOWED_HEADERS[sheetName];
  const allowedKeys = whitelist ? Object.keys(obj).filter(k => k && whitelist.includes(k)) : Object.keys(obj).filter(Boolean);
  const missingKeys = allowedKeys.filter(k => !headers.includes(k));
  if (missingKeys.length > 0) {
    const startCol = headers.length + 1;
    sh.getRange(1, startCol, 1, missingKeys.length).setValues([missingKeys]);
    headers = headers.concat(missingKeys);
  }
  return headers;
}

function sanitizeCellForSheet(val) {
  if (val === null || val === undefined) return '';
  if (typeof val === 'string') {
    const trimmed = val.trim();
    if (trimmed.length > 0 && ('=+-\t\r@'.indexOf(trimmed.charAt(0)) !== -1)) {
      return "'" + val; // Neutralize formula execution in Google Sheets
    }
  }
  return val;
}

/** Input Validation Helpers — Anti-DoS array size limits and NaN prevention */
function validateArraySize(arr, maxSize, label) {
  if (!Array.isArray(arr)) return [];
  if (arr.length > maxSize) {
    throw new Error('تجاوز الحد الأقصى المسموح لعناصر ' + label + ' (' + maxSize + ' عنصر)');
  }
  return arr;
}
function safeNumber(val, defaultVal) {
  const n = Number(val);
  return Number.isFinite(n) ? n : (defaultVal || 0);
}

function appendRow(sheetName, obj) {
  const sh = getSheet(sheetName);
  const headers = ensureHeaders(sh, obj);
  const row = headers.map(h => (obj[h] !== undefined ? sanitizeCellForSheet(obj[h]) : ''));
  sh.appendRow(row);
}

function upsertRow(sheetName, obj, keyField) {
  const sh = getSheet(sheetName);
  const headers = ensureHeaders(sh, obj);
  const keyCol = headers.indexOf(keyField);
  const values = sh.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][keyCol]) === String(obj[keyField])) { rowIndex = i; break; }
  }
  const row = headers.map(h => (obj[h] !== undefined ? sanitizeCellForSheet(obj[h]) : ''));
  if (rowIndex === -1) {
    sh.appendRow(row);
  } else {
    sh.getRange(rowIndex + 1, 1, 1, row.length).setValues([row]);
  }
}

function deleteRow(sheetName, keyValue, keyField) {
  const sh = getSheet(sheetName);
  const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  const keyCol = headers.indexOf(keyField);
  const values = sh.getDataRange().getValues();
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][keyCol]) === String(keyValue)) { sh.deleteRow(i + 1); break; }
  }
}

function logAudit(receiptId, user, action) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName('AuditLog');
  if (!sh) {
    sh = ss.insertSheet('AuditLog');
    sh.appendRow(['Timestamp', 'User', 'Action', 'ReceiptID']);
  }
  sh.appendRow([new Date(), user, action, receiptId]);
}

function jsonOut(obj, statusCode) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
